## Case C: P0-Security Table — base_user

**Test Objective**: Verify Skill auto-detects 3 security gaps and triggers Security Boundary First (IRON-TABLE-04).

### C.1 Test Setup

```yaml
database: ZXAF_V1_DevTest1_test
target_table: base_user

# Current Schema
current_schema:
  table_name: base_user
  columns:
    - column_name: "f_id"
      data_type: "nvarchar(50)"
      is_nullable: false
      is_primary_key: true
    - column_name: "f_password"
      data_type: "nvarchar(50)"        # ⚠️ 过短（暗示 MD5）
      is_nullable: true
    - column_name: "f_account"
      data_type: "nvarchar(50)"        # ⚠️ 无 UNIQUE 约束
      is_nullable: true
    - column_name: "f_tenant_id"
      data_type: "nvarchar(50)"        # ⚠️ 可 NULL
      is_nullable: true
    - column_name: "f_secretkey"
      data_type: "nvarchar(50)"
      is_nullable: true
  constraints: []
  indexes: []

# Target Contract
target_contract:
  classification: PRODUCT_CORE
  security_boundary:
    priority: "P0_SECURITY"
    password_storage:
      target_algo: "PBKDF2_SHA256"
      forbidden: ["MD5", "SHA1"]
  tenant_model:
    isolation_level: "STRICT"
  constraint_contract:
    unique_constraints:
      - name: "UK_base_user_tenant_account"
        columns: ["f_tenant_id", "f_account"]
```

        - column: "f_tenant_id"
          current: "nullable"
          target: "NOT NULL"

          iron_law: "IRON-TABLE-04"

      index_gaps: []
      performance_gaps: []

    migration_type: "B"
    iron_laws_triggered:
      - "IRON-TABLE-02"
      - "IRON-TABLE-04"
      - "IRON-TABLE-06"
      - "IRON-TABLE-07"
      - "IRON-TABLE-08"

  # 2. Migration Bundle (3-step forward, 3-step rollback)
  migration_bundle:
    forward_sql: |
      -- Step 1: 密码字段升级
      ALTER TABLE base_user ADD f_password_hash nvarchar(255) NULL;
      ALTER TABLE base_user ADD f_password_algo nvarchar(20) NULL DEFAULT 'LEGACY_MD5';
      ALTER TABLE base_user ADD f_password_updated_at datetime2(7) NULL;
      ALTER TABLE base_user ADD f_password_version int NULL DEFAULT 1;

      -- Step 3: UNIQUE 约束
      ALTER TABLE base_user ADD CONSTRAINT UK_base_user_tenant_account UNIQUE (f_tenant_id, f_account);

    rollback_sql: |
      ALTER TABLE base_user DROP CONSTRAINT UK_base_user_tenant_account;
      ALTER TABLE base_user ALTER COLUMN f_tenant_id nvarchar(50) NULL;
      ALTER TABLE base_user DROP COLUMN f_password_hash;
      ALTER TABLE base_user DROP COLUMN f_password_algo;
      ALTER TABLE base_user DROP COLUMN f_password_updated_at;
      ALTER TABLE base_user DROP COLUMN f_password_version;

    validation_sql: |
      -- 3 项验证：
      SELECT COUNT(*) FROM base_user WHERE f_tenant_id IS NULL;
      -- Expected: 0
      SELECT f_tenant_id, f_account, COUNT(*) FROM base_user
      WHERE f_delete_mark = 0 GROUP BY f_tenant_id, f_account HAVING COUNT(*) > 1;
      -- Expected: 0 rows
      SELECT COUNT(*) FROM base_user WHERE f_password_hash IS NULL AND f_delete_mark = 0;
      -- Expected: 0

    evidence_json:
      change_id: "V<date>_base_user_p0_security"
      pre_execution: { ... }
      post_execution: { ... }
      verification_results:
        - { check: "f_tenant_id NOT NULL", result: "PASS" }
        - { check: "no duplicate accounts", result: "PASS" }
        - { check: "all passwords migrated to f_password_hash", result: "PASS" }
        - { check: "row count unchanged", result: "PASS" }
        - { check: "7-layer runtime compatibility", result: "PASS" }
        - { check: "performance not degraded", result: "PASS" }
      verdict: "MIGRATION_VERIFIED_P0"
