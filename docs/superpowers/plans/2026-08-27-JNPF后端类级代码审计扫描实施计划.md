- [ ] **Step 1: 创建主扫描器框架**

```javascript
#!/usr/bin/env node

/**
 * JNPF 后端类级代码审计扫描器
 * 
 * 使用方法: node code-audit-scanner.mjs [--batch <batch-id>] [--module <module-name>] [--dimension <dimension-id>]
 */

import { readFile, writeFile, mkdir, readdir, stat } from 'fs/promises';
import { join, relative, basename } from 'path';
import { glob } from 'glob';

const CONFIG_PATH = '../scan-config.json';

async function loadConfig() {
  const configContent = await readFile(new URL(CONFIG_PATH, import.meta.url), 'utf-8');
  return JSON.parse(configContent);
}

async function collectFiles(modulesRoot, modulePath, exclusions) {
  const moduleDir = join(modulesRoot, modulePath);
  const pattern = join(moduleDir, '**', '*.cs');
  
  const files = await glob(pattern, {
    ignore: exclusions.map(e => join(modulesRoot, e.replace('**/', '')))
  });
  
  return files.filter(f => !f.includes('\\obj\\'));
}

async function scanFile(filePath, dimensions) {
  const content = await readFile(filePath, 'utf-8');
  const lines = content.split('\n');
  const findings = [];
  
  for (const dimension of dimensions) {
    for (const rule of dimension.rules) {
      const regex = new RegExp(rule.pattern, 'gm');
      let match;
      
      while ((match = regex.exec(content)) !== null) {
        const lineNumber = content.substring(0, match.index).split('\n').length;
        const lineContent = lines[lineNumber - 1]?.trim() || '';
        
        findings.push({
          dimension: dimension.id,
          rule: rule.id,
          severity: rule.severity,
          file: filePath,
          line: lineNumber,
          code: lineContent.substring(0, 100),
          description: rule.description,
          fix: rule.fix
        });
      }
    }
  }
  
  return findings;
}

async function scanModule(config, moduleName) {
  const module = config.modules.find(m => m.name === moduleName);
  if (!module) {
    console.error(`Module ${moduleName} not found in config`);
    return [];
  }
  
  console.log(`Scanning module: ${moduleName} (${module.estimatedFiles} files)`);
  
  const files = await collectFiles(config.modularityRoot, module.path, config.exclusions);
  console.log(`  Found ${files.length} files`);
  
  const allFindings = [];
  for (const file of files) {
    const findings = await scanFile(file, config.dimensions);
    allFindings.push(...findings);
  }
  
  console.log(`  Found ${allFindings.length} issues`);
  return allFindings;
}

async function generateReport(findings, config) {
  const report = {
    scanDate: config.scanDate,
    totalFiles: findings.length > 0 ? [...new Set(findings.map(f => f.file))].length : 0,
    totalIssues: findings.length,
    byDimension: {},
    bySeverity: { P0: 0, P1: 0, P2: 0, P3: 0 },
    topIssues: []
  };
  
  for (const finding of findings) {
    if (!report.byDimension[finding.dimension]) {
      report.byDimension[finding.dimension] = 0;
    }
    report.byDimension[finding.dimension]++;
    report.bySeverity[finding.severity]++;
  }
  
  report.topIssues = findings
    .filter(f => f.severity === 'P0' || f.severity === 'P1')
    .slice(0, 20);
  
  return report;
}

async function main() {
  const args = process.argv.slice(2);
  const batchIndex = args.indexOf('--batch');
  const moduleIndex = args.indexOf('--module');
  const dimensionIndex = args.indexOf('--dimension');
  
  const batchId = batchIndex !== -1 ? parseInt(args[batchIndex + 1]) : null;
  const moduleName = moduleIndex !== -1 ? args[moduleIndex + 1] : null;
  const dimensionId = dimensionIndex !== -1 ? args[dimensionIndex + 1] : null;
  
  const config = await loadConfig();
  
  let modulesToScan = [];
  
  if (moduleName) {
    modulesToScan = [moduleName];
  } else if (batchId) {
    const batch = config.batches.find(b => b.id === batchId);
    if (batch) {
      modulesToScan = batch.modules;
    }
  } else {
    modulesToScan = config.modules.map(m => m.name);
  }
  
  console.log(`Scanning ${modulesToScan.length} modules...`);
  
  const allFindings = [];
  
  for (const module of modulesToScan) {
    const findings = await scanModule(config, module);
    allFindings.push(...findings);
  }
  
  const report = await generateReport(allFindings, config);
  
  console.log('\n=== Scan Summary ===');
  console.log(`Total files scanned: ${report.totalFiles}`);
  console.log(`Total issues found: ${report.totalIssues}`);
  console.log('Issues by dimension:', report.byDimension);
  console.log('Issues by severity:', report.bySeverity);
  
  const outputPath = join(config.outputRoot, 'scan-summary.json');
  await writeFile(outputPath, JSON.stringify(report, null, 2));
  console.log(`\nReport saved to: ${outputPath}`);
}

main().catch(console.error);
```

```javascript
#!/usr/bin/env node

/**
 * 审计报告生成器
 */

import { readFile, writeFile, readdir } from 'fs/promises';
import { join } from 'path';

async function generateMarkdownReport(summary, findings, outputDir) {
  let report = `# JNPF 后端类级代码审计报告 v1.0

> **扫描日期**：${summary.scanDate}
> **扫描范围**：JNPF v5.2 后端全量业务代码
> **总文件数**：${summary.totalFiles}
> **发现问题**：${summary.totalIssues}

---

## 扫描摘要

| 维度 | 🔴 致命 | 🟠 严重 | 🟡 警告 | 🔵 建议 | 合计 |
|------|--------|--------|--------|--------|------|
`;

  const dimensions = ['A', 'B', 'C', 'D', 'E', 'F', 'G', 'H', 'I', 'J', 'K', 'L', 'M', 'N', 'O'];
  const dimensionNames = {
    A: '资源生命周期与内存泄漏',
    B: 'CLR 内存模型与 GC 压力',
    C: '异步编程模型',
    D: '线程安全与并发',
    E: '异常处理体系',
    F: '高绩效热路径',
    G: 'C# 现代特性缺失',
    H: '开闭原则与扩展性',
    I: '整洁架构与依赖方向',
    J: '安全与健壮性',
    K: '可观测性与诊断',
    L: '设计模式与代码结构',
    M: '日志与审计',
    N: 'JNPF 架构合规（铁律）',
    O: 'SqlSugar 安全与性能'
  };

  for (const dim of dimensions) {
    const dimFindings = findings.filter(f => f.dimension === dim);
    const p0 = dimFindings.filter(f => f.severity === 'P0').length;
    const p1 = dimFindings.filter(f => f.severity === 'P1').length;
    const p2 = dimFindings.filter(f => f.severity === 'P2').length;
    const p3 = dimFindings.filter(f => f.severity === 'P3').length;
    const total = dimFindings.length;
    
    report += `| ${dim} ${dimensionNames[dim]} | ${p0} | ${p1} | ${p2} | ${p3} | ${total} |\n`;
  }

  report += `| **合计** | ${summary.bySeverity.P0} | ${summary.bySeverity.P1} | ${summary.bySeverity.P2} | ${summary.bySeverity.P3} | ${summary.totalIssues} |\n`;

  report += `
---

## Top 20 最严重问题

| 排名 | 严重级别 | 维度 | 文件路径:行号 | 问题摘要 |
|------|---------|------|-------------|---------|
`;

  const topIssues = summary.topIssues.slice(0, 20);
  topIssues.forEach((issue, index) => {
    report += `| ${index + 1} | ${issue.severity} | ${issue.dimension} | ${issue.file}:${issue.line} | ${issue.description} |\n`;
  });

  report += `
---

## 维度详细分析

`;

  for (const dim of dimensions) {
    const dimFindings = findings.filter(f => f.dimension === dim);
    if (dimFindings.length === 0) continue;

    report += `### 维度 ${dim}：${dimensionNames[dim]}\n\n`;
    report += `| # | 严重级别 | 文件路径 | 行号 | 问题描述 | 修复建议 |\n`;
    report += `|---|---------|---------|------|---------|----------|\n`;

    dimFindings.slice(0, 10).forEach((finding, index) => {
      report += `| ${dim}-${String(index + 1).padStart(3, '0')} | ${finding.severity} | ${finding.file} | ${finding.line} | ${finding.description} | ${finding.fix} |\n`;
    });

    if (dimFindings.length > 10) {
      report += `\n> 共 ${dimFindings.length} 个问题，仅显示前 10 个\n\n`;
    }
  }

  report += `
---

## 修复收益预估

| 指标 | 当前状态 | 修复后目标 | 量化证据 |
|------|---------|-----------|---------|
| 多租户泄漏点 | ${findings.filter(f => f.dimension === 'N' && f.rule === 'N1').length} 处 | 0 处 | ITenantFilter 覆盖率 100% |
| SQL 注入风险 | ${findings.filter(f => f.dimension === 'J' && f.rule === 'J1').length} 处 | 0 处 | 全参数化 |
| API 权限缺失 | ${findings.filter(f => f.dimension === 'N' && f.rule === 'N3').length} 处 | 0 处 | SecurityDefine 覆盖率 100% |
| 内存泄漏点 | ${findings.filter(f => f.dimension === 'A').length} 处 | 0 处 | 长时间运行内存曲线持平 |
| 异常吞没 | ${findings.filter(f => f.dimension === 'E' && (f.rule === 'E1' || f.rule === 'E2')).length} 处 | 0 处 | Oops 体系覆盖率 100% |
| 架构违规 | ${findings.filter(f => f.dimension === 'I' || f.dimension === 'N').length} 处 | 0 处 | 架构测试通过 |

---

*报告生成时间：${new Date().toISOString()}*
`;

  return report;
}

async function main() {
  const configPath = join(process.cwd(), '.claude/evidence/backend-code-audit/scan-config.json');
  const summaryPath = join(process.cwd(), '.claude/evidence/backend-code-audit/scan-summary.json');
  const outputDir = join(process.cwd(), '.claude/evidence/backend-code-audit/reports');
  
  try {
    const summaryContent = await readFile(summaryPath, 'utf-8');
    const summary = JSON.parse(summaryContent);
    
    // 收集所有发现
    const findings = [];
    const modulesDir = join(process.cwd(), '.claude/evidence/backend-code-audit/modules');
    
    try {
      const modules = await readdir(modulesDir);
      for (const module of modules) {
        const moduleDir = join(modulesDir, module);
        const files = await readdir(moduleDir);
        for (const file of files) {
          if (file.startsWith('dimension-') && file.endsWith('.json')) {
            const content = await readFile(join(moduleDir, file), 'utf-8');
            const moduleFindings = JSON.parse(content);
            findings.push(...moduleFindings);
          }
        }
      }
    } catch (e) {
      console.log('No module findings found, using summary only');
    }
    
    // 生成 Markdown 报告
    const markdownReport = await generateMarkdownReport(summary, findings, outputDir);
    const markdownPath = join(outputDir, 'audit-report-v1.0.md');
    await writeFile(markdownPath, markdownReport);
    console.log(`Markdown report saved to: ${markdownPath}`);
    
    // 保存 Top 20 问题
    const top20Path = join(outputDir, 'top-20-issues.json');
    await writeFile(top20Path, JSON.stringify(summary.topIssues, null, 2));
    console.log(`Top 20 issues saved to: ${top20Path}`);
    
  } catch (error) {
    console.error('Error generating report:', error);
  }
}

main();
```
