-- backend/database/baseline/structure-snapshot.sql
-- 用途：导出 289 张表的完整结构（表/列/索引/约束/外键/CHECK）
-- 执行：sqlcmd -S (local)\SQLEXPRESS -d ZXAF_V1_DevTest1 -i structure-snapshot.sql

SET NOCOUNT ON;

-- 表 + 列
SELECT
    t.TABLE_NAME,
    c.COLUMN_NAME,
    c.DATA_TYPE,
    c.CHARACTER_MAXIMUM_LENGTH,
    c.IS_NULLABLE,
    c.COLUMN_DEFAULT,
    c.ORDINAL_POSITION
FROM INFORMATION_SCHEMA.TABLES t
JOIN INFORMATION_SCHEMA.COLUMNS c ON t.TABLE_NAME = c.TABLE_NAME
WHERE t.TABLE_SCHEMA = 'dbo'
ORDER BY t.TABLE_NAME, c.ORDINAL_POSITION;

-- 索引
SELECT
    OBJECT_NAME(i.object_id) AS table_name,
    i.name AS index_name,
    i.type_desc,
    i.is_unique,
    i.is_primary_key,
    STUFF((
        SELECT ',' + c.name + (CASE WHEN ic.is_descending_key = 1 THEN ' DESC' ELSE '' END)
        FROM sys.index_columns ic
        JOIN sys.columns c ON ic.object_id = c.object_id AND ic.column_id = c.column_id
        WHERE ic.object_id = i.object_id AND ic.index_id = i.index_id
        ORDER BY ic.key_ordinal
        FOR XML PATH('')
    ), 1, 1, '') AS columns,
    i.filter_definition
FROM sys.indexes i
WHERE OBJECT_SCHEMA_NAME(i.object_id) = 'dbo'
ORDER BY table_name, index_name;

-- 外键
SELECT
    OBJECT_NAME(fk.parent_object_id) AS table_name,
    fk.name AS fk_name,
    OBJECT_NAME(fk.referenced_object_id) AS referenced_table,
    COL_NAME(fkc.parent_object_id, fkc.parent_column_id) AS column_name,
    COL_NAME(fkc.referenced_object_id, fkc.referenced_column_id) AS referenced_column
FROM sys.foreign_keys fk
JOIN sys.foreign_key_columns fkc ON fk.object_id = fkc.constraint_object_id
ORDER BY table_name, fk_name;

-- CHECK 约束
SELECT
    OBJECT_NAME(parent_object_id) AS table_name,
    name AS check_name,
    definition
FROM sys.check_constraints
ORDER BY table_name, check_name;
```

# scripts/lib/scan-repository-callgraph.ps1
# 用途：扫描 modularity/**/Repository.cs，提取 IRepository<T> 调用图

param(
    [string]$OutputPath = "backend/database/baseline/repository-callgraph.json"
)

$pattern = "IRepository<\w+>\.\w+\(.*?\)"
$files = Get-ChildItem -Path "backend/modularity" -Filter "*Repository.cs" -Recurse

$callgraph = @()
foreach ($file in $files) {
    $content = Get-Content $file.FullName -Raw
    $matches_found = [regex]::Matches($content, $pattern)
    foreach ($m in $matches_found) {
        $callgraph += @{
            File = $file.FullName.Replace((Get-Location).Path + "\", "")
            Line = ($content.Substring(0, $m.Index) -split "`n").Count
            Call = $m.Value
        }
    }
}

$callgraph | ConvertTo-Json -Depth 3 | Set-Content -Path $OutputPath -Encoding UTF8
Write-Host "Scan complete: $($callgraph.Count) IRepository calls in $($files.Count) files"
Write-Host "Output: $OutputPath"
```

-- 用途：导出 289 张表的 COUNT(*) 作为行数基线
-- 输出：backend/database/baseline/row-count-snapshot.json

SET NOCOUNT ON;

DECLARE @result NVARCHAR(MAX) = '{';
DECLARE @first BIT = 1;

DECLARE cur CURSOR LOCAL FAST_FORWARD FOR
SELECT TABLE_NAME FROM INFORMATION_SCHEMA.TABLES WHERE TABLE_SCHEMA = 'dbo' ORDER BY TABLE_NAME;

DECLARE @table NVARCHAR(255);
OPEN cur;
FETCH NEXT FROM cur INTO @table;

WHILE @@FETCH_STATUS = 0
BEGIN
    DECLARE @sql NVARCHAR(MAX) = N'SELECT @cnt = COUNT(*) FROM dbo.' + QUOTENAME(@table);
    DECLARE @cnt BIGINT;
    EXEC sp_executesql @sql, N'@cnt BIGINT OUTPUT', @cnt OUTPUT;

    IF @first = 0 SET @result = @result + ',';
    SET @result = @result + '"' + @table + '":' + CAST(@cnt AS NVARCHAR(20));
    SET @first = 0;

    FETCH NEXT FROM cur INTO @table;
END

CLOSE cur;
DEALLOCATE cur;

SET @result = @result + '}';
PRINT @result;
```

-- scripts/database-modernization/phase1-p0-security/01-add-password-fields.sql
-- 阶段 1.1：在 base_user 上新增密码相关字段（保留旧 f_password 字段，6 个月兼容期）

BEGIN TRAN;

IF NOT EXISTS (SELECT 1 FROM sys.columns WHERE object_id = OBJECT_ID('base_user') AND name = 'f_password_hash')
    ALTER TABLE base_user ADD f_password_hash nvarchar(255) NULL;

IF NOT EXISTS (SELECT 1 FROM sys.columns WHERE object_id = OBJECT_ID('base_user') AND name = 'f_password_algo')
    ALTER TABLE base_user ADD f_password_algo nvarchar(20) NULL DEFAULT 'LEGACY_MD5';

IF NOT EXISTS (SELECT 1 FROM sys.columns WHERE object_id = OBJECT_ID('base_user') AND name = 'f_password_updated_at')
    ALTER TABLE base_user ADD f_password_updated_at datetime2(7) NULL;

IF NOT EXISTS (SELECT 1 FROM sys.columns WHERE object_id = OBJECT_ID('base_user') AND name = 'f_password_version')
    ALTER TABLE base_user ADD f_password_version int NULL DEFAULT 1;

-- 数据回填：新字段用旧字段值
UPDATE base_user
SET f_password_hash = f_password,
    f_password_algo = 'LEGACY_MD5',
    f_password_updated_at = ISNULL(f_change_password_date, SYSUTCDATETIME()),
    f_password_version = 1
WHERE f_password IS NOT NULL AND f_password_hash IS NULL;

COMMIT;
GO
```
