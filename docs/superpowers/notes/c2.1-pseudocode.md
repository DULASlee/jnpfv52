## 2. 适配器核心逻辑（伪代码）

```csharp
public class IrToVisualDevAdapter
{
    private readonly ISqlSugarClient _db;

    /// <summary>
    /// 将 IR JSON 转换为临时 VisualDevEntity 记录并落库.
    /// 返回临时记录 ID（供 CodeGenService 使用）.
    /// </summary>
    public async Task<string> ConvertAndCreateAsync(
        string irJson, string tenantId, long userId)
    {
        // 1. 反序列化 IR
        var ir = JsonSerializer.Deserialize<IntermediateRepresentation>(irJson);
        if (ir == null || ir.Tables.Count == 0)
            throw new InvalidOperationException("IR 为空或无表定义");

        // 2. 构造 FormData JSON（CodeGenService 需要）
        var formData = BuildFormData(ir);

        // 3. 构造 Tables JSON（表关系）
        var tablesJson = BuildTablesJson(ir);

        // 4. 构造 ColumnData（从字段列表自动派生列设计）
        var columnData = BuildColumnData(ir);

        // 5. 创建临时 VisualDevEntity 记录
        var entity = new VisualDevEntity
        {
            Id = Guid.NewGuid().ToString("N"),
            FullName = $"[AI-TEMP] {ir.Modules.FirstOrDefault()?.ModuleName ?? "未命名"}",
            FormData = formData.ToJsonString(),
            Tables = tablesJson.ToJsonString(),
            ColumnData = columnData?.ToJsonString(),
            Category = await GetDefaultCategoryAsync(),
            WebType = 1,  // PC
            Type = 1,     // 标准模板
            DeleteMark = null,
            // 审计字段继承自 TenantCLDSEntityBase
            CreatorTime = DateTime.Now,
            CreatorUserId = userId,
            TenantId = tenantId
        };

        await _db.Insertable(entity).ExecuteCommandAsync();
        return entity.Id;
    }

    /// <summary>
    /// 从 IR 的 tables[].fields[] → FormData（formField 列表）.
    /// </summary>
    private FormDataModel BuildFormData(IntermediateRepresentation ir)
    {
        var firstTable = ir.Tables.First();
        var model = new FormDataModel
        {
            // className 从表名提取（PascalCase）
            className = ir.Tables.Select(t => t.TableName.ParseToPascalCase()).ToList(),
            // areasName 从第一个模块名提取
            areasName = ir.Modules.FirstOrDefault()?.ModuleName ?? "Default",
            // formField 从第一个表的字段列表提取
            formField = firstTable.Fields
                .Where(f => !f.IsPrimaryKey) // 主键不出现在表单中
                .Select(f => new FormFieldModel
                {
                    fieldName = f.FieldName,
                    fieldType = f.FieldType,
                    fieldLength = f.FieldLength
                })
                .ToList()
        };
        return model;
    }

    /// <summary>
    /// 从 IR 的 tables[] + relations[] → Tables JSON（DbTableRelationModel 列表）.
    /// </summary>
    private List<DbTableRelationModel> BuildTablesJson(IntermediateRepresentation ir)
    {
        return ir.Tables.Select(table => new DbTableRelationModel
        {
            TableName = table.TableName,
            // 子表信息从 relations 中提取
            RelationField = ir.Relations?
                .Where(r => r.FromTable == table.TableName)
                .Select(r => r.FromField)
                .FirstOrDefault() ?? ""
        }).ToList();
    }

    /// <summary>
    /// 从 IR 的 tables[].fields[] → ColumnData（列表页列设计）.
    /// 降级策略：取前 4 个非主键字段，每列宽 150px.
    /// </summary>
    private ColumnDesignModel? BuildColumnData(IntermediateRepresentation ir)
    {
        var firstTable = ir.Tables.FirstOrDefault();
        if (firstTable == null) return null;

        return new ColumnDesignModel
        {
            type = 1, // 标准列
            columns = firstTable.Fields
                .Where(f => !f.IsPrimaryKey)
                .Take(4)
                .Select(f => new ColumnModel
                {
                    prop = f.FieldName,
                    label = f.FieldName, // 降级：字段名即列名
                    width = 150,
                    // 字典字段用 jnpf-select 组件
                    isDictionary = f.IsDictionary,
                    dictionaryCode = f.DictionaryCode
                })
                .ToList()
        };
    }
}
```

## 3. CodeGen 调用器（精确版 — 基于源码验证）

### 关键约束
- `CodeGenService` 是 `ITransient`，DI 可注入
- `DownloadCode` 输出路径硬编码 → 生成后需移动文件到 workspace
- `CLDEntityBase.Creator()` 访问 `App.User` → 在 API 端点上下文中天然可用

```csharp
/// <summary>
/// C2.1 CodeGen 调用器 — DI 注入 CodeGenService，生成后移动文件到 workspace.
/// </summary>
public class CodeGenCaller : ITransient
{
    private readonly CodeGenService _codeGenService;       // DI 注入
    private readonly ISqlSugarClient _db;
    private readonly ILogger<CodeGenCaller> _logger;

    public CodeGenCaller(
        CodeGenService codeGenService,   // ← 直接注入，不要 HTTP 代理
        ISqlSugarClient db,
        ILogger<CodeGenCaller> logger)
    {
        _codeGenService = codeGenService;
        _db = db;
        _logger = logger;
    }

    /// <summary>
    /// 从 IR JSON 生成完整代码，产物落入 StudioWorkspace/{tenantId}/{pipelineId}/generated/.
    /// 返回 generated/ 目录路径.
    /// </summary>
    public async Task<string> GenerateFromIrAsync(
        string irJson, string tenantId, string pipelineId, long userId)
    {
        var adapter = new IrToVisualDevAdapter(_db);
        string? tempEntityId = null;

        try
        {
            // 1. IR JSON → 临时 VisualDevEntity 记录
            tempEntityId = await adapter.ConvertAndCreateAsync(irJson, tenantId, userId);
            _logger.LogInformation("临时 VisualDev 记录已创建: {Id}", tempEntityId);

            // 2. 构造 DownloadCodeFormInput
            var ir = JsonSerializer.Deserialize<IntermediateRepresentation>(irJson);
            var mainTable = ir!.Tables.First();
            var input = new DownloadCodeFormInput
            {
                module = await ResolveModuleIdAsync(ir.Modules.First().ModuleName),
                className = mainTable.TableName,             // ← 原始表名，DownloadCode 内部做 PascalCase
                subClassName = string.Join(",", ir.Tables.Skip(1).Select(t => t.TableName)),
                description = mainTable.TableName
            };

            // 3. 调用 CodeGenService — 产物落在 CodeGenerate/{fileName}/
            var result = await _codeGenService.DownloadCode(tempEntityId, input);
            var codegenDir = InferCodeGeneratePath(result);  // 从返回值反推路径
            _logger.LogInformation("代码生成完成: {Path}", codegenDir);

            // 4. 移动到 AI 工作区
            var (_, generatedDir, _, _) = StudioWorkspaceHelper
                .GetPipelineSubPaths(tenantId, pipelineId);
            CopyDirectoryRecursive(codegenDir, generatedDir);
            _logger.LogInformation("产物已移动到工作区: {Dir}", generatedDir);

            // 5. 可选：清理 CodeGenerate 临时文件
            try { Directory.Delete(codegenDir, true); } catch { /* ignore */ }

            return generatedDir;
        }
        finally
        {
            // 6. 清理临时 VisualDevEntity 记录
            if (tempEntityId != null)
            {
                await _db.Deleteable<VisualDevEntity>()
                    .Where(e => e.Id == tempEntityId)
                    .ExecuteCommandAsync();
                _logger.LogInformation("临时记录已清理: {Id}", tempEntityId);
            }
        }
    }
}
```

## 4. 模糊点查清结果（2026-07-03 源码探索）

### 模糊点 1：CodeGenService 调用方式 ✅ RESOLVED

**源码位置**: `backend/modularity/codegen/JNPF.CodeGen/CodeGenService.cs:38`

```csharp

// DownloadCode 签名
[HttpPost("{id}/Actions/DownloadCode")]
public async Task<dynamic> DownloadCode(string id, [FromBody] DownloadCodeFormInput downloadCodeForm)
// id = VisualDevEntity.Id (string)
// 返回值 = { name: string, url: string }  — zip 下载 URL
```

**关键发现**: 输出路径**硬编码**在第 160 行：

### 模糊点 3：Tables JSON 精确格式 ✅ RESOLVED

**Tables** = `List<DbTableRelationModel>` 序列化 JSON：

```json
[
  {
    "typeId": "1",        // "1"=主表, "0"=子表
    "table": "Student",   // 数据库表名
    "tableName": "学生表", // 中文说明
    "tableKey": "id",     // 主键字段
    "tableField": "",     // 外键字段（子表用）
    "relationTable": "",  // 关联主表名（子表用）
    "relationField": "",  // 关联主表主键（子表用）
    "className": "StudentInfo" // PascalCase 类名
  }
]
```

**FormData** = `FormDataModel` 序列化 JSON：

```json
{
  "areasName": "学生管理",
  "className": ["StudentInfo"],
  "fields": [
    {
      "__config__": { "label": "姓名", "jnpfKey": "name" },
      "__vModel__": "name",
      "placeholder": "请输入姓名",
      "maxlength": 50
    }
  ],
  "primaryKeyPolicy": 1
}
```

最小可行 FormData（只需 `areasName` + `className` + `fields`）:
- `fields[].__config__.label` ← 从 `IrField.FieldName` 派生
- `fields[].__vModel__` ← 从 `IrField.FieldName` 派生
- `fields[].maxlength` ← 从 `IrField.FieldLength` 派生

### 模糊点 4：临时记录清理方式 ✅ RESOLVED

**可用基础设施**:
1. **Hangfire**: `BackgroundJob` 已在项目中使用 (`PipelineValidationJob`)
2. **IHostedService**: `BackgroundTaskShutdownService` 已注册
3. **同步清理**: finally 块中直接删除

**推荐方案**: 
- **成功/失败路径**: finally 块硬删除 `_repository.DeleteAsync(tempEntityId)`（同步，0 延迟）
- **兜底清理**: Hangfire `RecurringJob` 每日清扫 24h+ 的孤立记录（防止进程崩溃遗留）
- **配置**: `StudioPreview:TempRecordCleanupHours` = 24

### 模糊点 5：HTTP Context 依赖 ✅ RESOLVED

**`DownloadCode` 无 HttpContext 依赖！** 该方法不访问 `App.HttpContext` 或 `HttpContext.User`。

但 `CLDEntityBase.Creator()` 访问 `App.User`（JWT Claims）。解决方法：
- 不调用 `entity.Creator()`，手动设置 `Id` + `CreatorTime` + `CreatorUserId`
- C2.1 在受认证的 API 端点内运行，`App.User` 自然可用

**调用链路确认**：C# DI 注入 `CodeGenService` → 直接调用 `DownloadCode(tempEntityId, input)` → ✅ 完全可行，不需要 HTTP 代理。
