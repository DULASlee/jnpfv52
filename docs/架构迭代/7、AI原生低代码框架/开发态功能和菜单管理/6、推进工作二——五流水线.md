```csharp
// ============================================================

// 描述: 大模型网关 - 统一对接 DeepSeek / 通义千问 / OpenAI

using JNPF.Application.Studio.Models;

    public string Provider { get; set; } = string.Empty;

/// <summary>
/// 大模型网关服务
/// </summary>
public class LlmGatewayService
{
    private readonly HttpClient _httpClient;
    private readonly ISqlSugarClient _db;
    private readonly LlmProviderConfig _config;
    private readonly ILogger<LlmGatewayService> _logger;

    public LlmGatewayService(
        IHttpClientFactory httpClientFactory,
        ISqlSugarClient db,
        IConfiguration configuration,
        ILogger<LlmGatewayService> logger)
    {
        _httpClient = httpClientFactory.CreateClient("LlmGateway");
        _db = db;
        _logger = logger;

        // 从配置文件读取 LLM 供应商配置
        _config = configuration.GetSection("LlmGateway").Get<LlmProviderConfig>()
            ?? new LlmProviderConfig
            {
                Provider = "deepseek",
                BaseUrl = "https://api.deepseek.com",
                Model = "deepseek-chat",
                ApiKey = ""
            };
    }

    /// <summary>
    /// 非流式对话
    /// </summary>
    public async Task<LlmChatResponse> ChatAsync(
        LlmChatRequest request,
        long? userId = null,
        string? tenantId = null,
        CancellationToken ct = default)
    {
        var start = DateTimeOffset.UtcNow;
        Exception? lastError = null;

        for (int attempt = 0; attempt < request.MaxRetries; attempt++)
        {
            try
            {
                var httpRequest = BuildHttpRequest(request);
                using var cts = CancellationTokenSource
                    .CreateLinkedTokenSource(ct);
                cts.CancelAfter(request.TimeoutMs);

                var httpResponse = await _httpClient
                    .SendAsync(httpRequest, HttpCompletionOption
                        .ResponseContentRead, cts.Token);
                httpResponse.EnsureSuccessStatusCode();

                var json = await httpResponse.Content
                    .ReadAsStringAsync(cts.Token);
                var result = ParseResponse(json);

                var latency = (DateTimeOffset.UtcNow - start)
                    .TotalMilliseconds;
                result.LatencyMs = (long)latency;
                result.Provider = _config.Provider;

                // 审计日志
                await WriteCallLog(result, userId, tenantId);

                return result;
            }
            catch (Exception ex)
            {
                lastError = ex;
                _logger.LogWarning(ex,
                    "LLM 调用失败，第 {Attempt} 次重试", attempt + 1);

                if (attempt < request.MaxRetries - 1)
                {
                    await Task.Delay(
                        (int)Math.Pow(2, attempt) * 1000, ct);
                }
            }
        }

        throw new InvalidOperationException(
            $"LLM 调用失败，已重试 {request.MaxRetries} 次: {lastError?.Message}",
            lastError);
    }

    /// <summary>
    /// 流式对话 (SSE) — 返回 IAsyncEnumerable，逐 chunk 推送
    /// </summary>
    public async IAsyncEnumerable<string> ChatStreamAsync(
        LlmChatRequest request,
        long? userId = null,
        string? tenantId = null,
        [EnumeratorCancellation] CancellationToken ct = default)
    {
        var start = DateTimeOffset.UtcNow;
        var httpRequest = BuildHttpRequest(request, stream: true);
        using var cts = CancellationTokenSource
            .CreateLinkedTokenSource(ct);
        cts.CancelAfter(request.TimeoutMs);

        var httpResponse = await _httpClient
            .SendAsync(httpRequest,
                HttpCompletionOption.ResponseHeadersRead, cts.Token);
        httpResponse.EnsureSuccessStatusCode();

        using var stream = await httpResponse.Content
            .ReadAsStreamAsync(cts.Token);
        using var reader = new StreamReader(stream, Encoding.UTF8);

        int totalCompletionTokens = 0;
        string? buffer = null;

        while (!reader.EndOfStream)
        {
            var line = await reader.ReadLineAsync();
            if (ct.IsCancellationRequested) break;

            if (string.IsNullOrWhiteSpace(line)) continue;

            if (!line.StartsWith("data: ")) continue;
            var data = line["data: ".Length..];

            if (data == "[DONE]")
            {
                // 流结束，写审计日志
                var latency = (long)(DateTimeOffset.UtcNow - start)
                    .TotalMilliseconds;
                await WriteCallLogAsync(new LlmChatResponse
                {
                    CompletionTokens = totalCompletionTokens,
                    Provider = _config.Provider,
                    Model = _config.Model,
                    LatencyMs = latency
                }, userId, tenantId);
                yield break;
            }

            string? chunk = null;
            try
            {
                using var doc = JsonDocument.Parse(data);
                var root = doc.RootElement;
                chunk = root
                    .GetProperty("choices")[0]
                    .GetProperty("delta")
                    .TryGetProperty("content", out var c)
                    ? c.GetString()
                    : null;
            }
            catch
            {
                // 忽略解析错误
            }

            if (!string.IsNullOrEmpty(chunk))
            {
                totalCompletionTokens++;
                yield return chunk;
            }
        }
    }

    #region 私有方法

    private HttpRequestMessage BuildHttpRequest(
        LlmChatRequest request, bool stream = false)
    {
        var messages = request.Messages.Select(m => new
        {
            role = m.Role,
            content = m.Content
        }).ToList();

        var body = new Dictionary<string, object>
        {
            ["model"] = _config.Model,
            ["messages"] = messages,
            ["max_tokens"] = _config.MaxTokens,
            ["temperature"] = _config.Temperature,
            ["stream"] = stream
        };

        if (request.ResponseFormat == "json")
        {
            body["response_format"] = new
            { type = "json_object" };
        }

        var json = JsonSerializer.Serialize(body);
        var httpRequest = new HttpRequestMessage(
            HttpMethod.Post,
            $"{_config.BaseUrl}/v1/chat/completions")
        {
            Content = new StringContent(json, Encoding.UTF8,
                "application/json")
        };

        httpRequest.Headers.Authorization =
            new AuthenticationHeaderValue("Bearer", _config.ApiKey);

        return httpRequest;
    }

    private LlmChatResponse ParseResponse(string json)
    {
        using var doc = JsonDocument.Parse(json);
        var root = doc.RootElement;

        var content = root
            .GetProperty("choices")[0]
            .GetProperty("message")
            .GetProperty("content")
            .GetString() ?? "";

        var usage = root.TryGetProperty("usage", out var u)
            ? u
            : default;

        return new LlmChatResponse
        {
            Content = content,
            PromptTokens = usage.ValueKind != JsonValueKind.Undefined
                && usage.TryGetProperty("prompt_tokens", out var pt)
                ? pt.GetInt32() : 0,
            CompletionTokens = usage.ValueKind != JsonValueKind.Undefined
                && usage.TryGetProperty("completion_tokens", out var ct2)
                ? ct2.GetInt32() : 0,
            TotalTokens = usage.ValueKind != JsonValueKind.Undefined
                && usage.TryGetProperty("total_tokens", out var tt)
                ? tt.GetInt32() : 0,
            Model = root.TryGetProperty("model", out var m)
                ? m.GetString() ?? _config.Model : _config.Model
        };
    }

    private async Task WriteCallLog(
        LlmChatResponse response,
        long? userId, string? tenantId)
    {
        try
        {
            await _db.Insertable(new
            {
                F_Id = SnowFlakeSingle.Instance.NextId(),
                F_TenantId = tenantId ?? "",
                F_UserId = userId ?? 0,
                F_Provider = response.Provider,
                F_Model = response.Model,
                F_PromptTokens = response.PromptTokens,
                F_CompletionTokens = response.CompletionTokens,
                F_TotalTokens = response.TotalTokens,
                F_Latency = (int)response.LatencyMs,
                F_Status = "success",
                F_CreatorTime = DateTime.Now
            }).AS("BASE_AI_CALL_LOG").ExecuteCommandAsync();
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "写入 AI 调用日志失败");
        }
    }

    private async Task WriteCallLogAsync(
        LlmChatResponse response,
        long? userId, string? tenantId)
    {
        try
        {
            await _db.Insertable(new
            {
                F_Id = SnowFlakeSingle.Instance.NextId(),
                F_TenantId = tenantId ?? "",
                F_UserId = userId ?? 0,
                F_Provider = response.Provider,
                F_Model = response.Model,
                F_CompletionTokens = response.CompletionTokens,
                F_Latency = (int)response.LatencyMs,
                F_Status = "success",
                F_CreatorTime = DateTime.Now
            }).AS("BASE_AI_CALL_LOG").ExecuteCommandAsync();
        }
        catch { }
    }

    #endregion
}
```

```csharp
[Authorize]
public class PromptTemplateService : IDynamicApiController
{
    private readonly ISqlSugarClient _db;

    public PromptTemplateService(ISqlSugarClient db)
    {
        _db = db;
    }

    /// <summary>
    /// 根据模板编码获取并渲染 Prompt
    /// </summary>
    public async Task<string> RenderPromptAsync(
        string templateCode,
        Dictionary<string, object> variables)
    {
        var template = await _db.Queryable<PromptTemplateEntity>()
            .Where(t => t.F_TemplateCode == templateCode
                     && t.F_Enabled && !t.F_DeleteMark)
            .FirstAsync()
            ?? throw new InvalidOperationException(
                $"Prompt模板 '{templateCode}' 不存在或已禁用");

        return Interpolate(template.F_SystemPrompt, variables);
    }

    /// <summary>
    /// 获取模板（不做渲染）
    /// </summary>
    public async Task<PromptTemplateEntity?> GetTemplateAsync(
        string templateCode)
    {
        return await _db.Queryable<PromptTemplateEntity>()
            .Where(t => t.F_TemplateCode == templateCode
                     && t.F_Enabled && !t.F_DeleteMark)
            .FirstAsync();
    }

    /// <summary>
    /// 变量插值：将 {{variableName}} 替换为实际值
    /// </summary>
    public static string Interpolate(
        string template,
        Dictionary<string, object> variables)
    {
        if (string.IsNullOrEmpty(template) || variables == null)
            return template;

        return Regex.Replace(template, @"\{\{(\w+)\}\}", match =>
        {
            var key = match.Groups[1].Value;
            if (variables.TryGetValue(key, out var value))
            {
                return value is string str
                    ? str
                    : JsonSerializer.Serialize(value,
                        new JsonSerializerOptions
                        { WriteIndented = true });
            }
            return match.Value; // 未找到变量，保留原始占位符
        });
    }

    #region CRUD

    [HttpGet("api/prompt-template/list")]
    public async Task<List<PromptTemplateEntity>> GetList()
    {
        return await _db.Queryable<PromptTemplateEntity>()
            .Where(t => !t.F_DeleteMark)
            .OrderBy(t => t.F_TemplateCode)
            .ToListAsync();
    }

    [HttpPost("api/prompt-template/save")]
    public async Task<long> Save(PromptTemplateEntity input)
    {
        if (input.F_Id == 0)
        {
            input.F_Id = SnowFlakeSingle.Instance.NextId();
            input.F_CreatorTime = DateTime.Now;
            await _db.Insertable(input).ExecuteCommandAsync();
        }
        else
        {
            input.F_ModifyTime = DateTime.Now;
            await _db.Updateable(input)
                .IgnoreColumns(t => new { t.F_CreatorTime, t.F_CreatorUserId })
                .ExecuteCommandAsync();
        }
        return input.F_Id;
    }

    #endregion
}
```

```csharp
    public async Task StreamMessage(
        long id,
        [FromQuery] string message,
        CancellationToken ct)
    {
        var httpContext = HttpContext;
        httpContext.Response.ContentType = "text/event-stream";
        httpContext.Response.Headers["Cache-Control"] = "no-cache";
        httpContext.Response.Headers["X-Accel-Buffering"] = "no";

        var pipeline = await GetPipelineOrThrow(id);

        async Task SendSse(string eventType, object data)
        {
            var json = JsonSerializer.Serialize(data,
                new JsonSerializerOptions
                { PropertyNamingPolicy = JsonNamingPolicy.CamelCase });
            await httpContext.Response.WriteAsync(
                $"event: {eventType}\ndata: {json}\n\n", ct);
            await httpContext.Response.Body.FlushAsync(ct);
        }

        try
        {
            // 存储用户消息
            if (!string.IsNullOrWhiteSpace(message))
            {
                await SaveMessage(id, pipeline.F_CurrentStage,
                    "user", message, "text");
            }

            // 根据当前阶段分发到对应处理器
            switch ((PipelineStage)pipeline.F_CurrentStage)
            {
                case PipelineStage.Requirement:
                    await _stage1.ProcessAsync(
                        pipeline, message, _llm, _promptService,
                        SendSse, ct);
                    break;

                case PipelineStage.Architecture:
                    await _stage2.ProcessAsync(
                        pipeline, message, _llm, _promptService,
                        SendSse, ct);
                    break;

                case PipelineStage.DetailedDesign:
                    await _stage3.ProcessAsync(
                        pipeline, message, _llm, _promptService,
                        SendSse, ct);
                    break;

                case PipelineStage.Development:
                    await _stage4.ProcessAsync(
                        pipeline, _llm, SendSse, ct);
                    break;

                case PipelineStage.Delivery:
                    await _stage5.ProcessAsync(
                        pipeline, SendSse, ct);
                    break;
            }
        }
        catch (Exception ex)
        {
            _logger.LogError(ex,
                "流水线 {PipelineId} 阶段 {Stage} 处理异常",
                id, pipeline.F_CurrentStage);

            await SendSse(SseEventType.Error, new
            {
                message = $"处理异常: {ex.Message}"
            });
        }
    }

    /// <summary>

    [HttpPost("api/ai/pipeline/{id}/confirm")]
    public async Task<object> ConfirmStage(
        long id, [FromBody] ConfirmStageInput input)
    {
        var pipeline = await GetPipelineOrThrow(id);

        if (pipeline.F_CurrentStage != input.Stage)
        {
            return new
            {
                success = false,
                message = "当前阶段不匹配"
            };
        }

        if (!input.Approved)
        {
            // 用户不满意，AI 需要根据反馈修改
            // 把反馈作为系统消息存储，下次 AI 处理时会读到
            await SaveMessage(id, input.Stage, "system",
                $"用户对阶段{input.Stage}的产出提出修改意见: {input.Feedback}",
                "text");

            // 重置状态为 active，让用户继续发送消息
            pipeline.F_Status = "active";
            pipeline.F_ModifyTime = DateTime.Now;
            await _db.Updateable(pipeline)
                .UpdateColumns(p => new
                { p.F_Status, p.F_ModifyTime })
                .ExecuteCommandAsync();

            return new
            {
                success = true,
                message = "已记录修改意见，请继续与 AI 交互",
                nextStage = input.Stage,
                status = "active"
            };
        }

        // 标记当前阶段消息为已确认
        await _db.Updateable<PipelineMessageEntity>()
            .SetColumns(m => m.F_IsConfirmed, true)
            .Where(m => m.F_PipelineId == id
                     && m.F_Stage == input.Stage)
            .ExecuteCommandAsync();

        // 推进到下一阶段
        if (input.Stage < 5)
        {
            var nextStage = input.Stage + 1;
            pipeline.F_CurrentStage = nextStage;
            pipeline.F_Status = "active";
            pipeline.F_ModifyTime = DateTime.Now;
            await _db.Updateable(pipeline)
                .UpdateColumns(p => new
                { p.F_CurrentStage, p.F_Status, p.F_ModifyTime })
                .ExecuteCommandAsync();

            // 插入系统消息，提示阶段推进
            await SaveMessage(id, nextStage, "system",
                $"阶段{input.Stage}已确认通过，进入阶段{nextStage}",
                "text");

            return new
            {
                success = true,
                message = $"已进入阶段 {nextStage}",
                nextStage,
                status = "active"
            };
        }
        else
        {
            // 阶段5确认 → 全部完成
            pipeline.F_Status = "completed";
            pipeline.F_ModifyTime = DateTime.Now;
            await _db.Updateable(pipeline)
                .UpdateColumns(p => new
                { p.F_Status, p.F_ModifyTime })
                .ExecuteCommandAsync();

            return new
            {
                success = true,
                message = "全部完成！系统已生成",
                nextStage = 5,
                status = "completed"
            };
        }
    }
```

    private const int MIN_QUESTIONS = 3; // 每轮至少 3 个问题

    /// <summary>
    /// 处理阶段一逻辑
    /// </summary>
    public async Task ProcessAsync(
        PipelineEntity pipeline,
        string userMessage,
        LlmGatewayService llm,
        PromptTemplateService promptService,
        Func<string, object, Task> sendSse,
        CancellationToken ct)
    {
        var pipelineId = pipeline.F_Id;

        await sendSse(SseEventType.Thinking, new
        {
            message = "正在分析您的需求..."
        });

        // 1. 获取行业知识上下文（如果租户配置了）
        var industryContext = await GetIndustryContext(
            promptService, pipeline.F_TenantId);

        // 2. 获取术语表上下文
        var glossaryContext = await GetGlossaryContext(
            promptService, pipeline.F_TenantId);

        // 3. 构建 System Prompt
        var systemPrompt = await BuildSystemPrompt(
            promptService, industryContext, glossaryContext);

        // 4. 构建消息列表
        var messages = await _pipelineService
            .BuildContextAsync(pipelineId, 1);

        // 加入系统 Prompt
        messages.Insert(0, new LlmMessage
        {
            Role = "system",
            Content = systemPrompt
        });

        // 5. 流式调用 LLM
        var fullResponse = new System.Text.StringBuilder();

        await foreach (var chunk in llm.ChatStreamAsync(
            new LlmChatRequest
            {
                Messages = messages,
                ResponseFormat = "json",
                MaxRetries = 2,
                TimeoutMs = 120000
            },
            pipeline.F_UserId,
            pipeline.F_TenantId,
            ct))
        {
            fullResponse.Append(chunk);

            // 实时推送文本片段给前端
            await sendSse(SseEventType.Chunk, new
            {
                content = chunk,
                stage = 1
            });
        }

        var responseText = fullResponse.ToString();

        // 6. 解析 AI 响应
        RequirementAnalysisResult? analysis = null;
        try
        {
            // 尝试提取 JSON
            var jsonStr = ExtractJson(responseText);
            analysis = JsonSerializer
                .Deserialize<RequirementAnalysisResult>(
                    jsonStr,
                    new JsonSerializerOptions
                    {
                        PropertyNameCaseInsensitive = true
                    });
        }
        catch (Exception ex)
        {
            _logger.LogWarning(ex, "解析需求分析响应失败，按纯文本处理");
        }

        // 7. 保存 AI 响应消息
        await _pipelineService.SaveMessage(
            pipelineId, 1, "assistant",
            responseText, "json");

        // 8. 判断是否需要继续追问
        int currentRound = _roundCounter
            .GetValueOrDefault(pipelineId, 0);

        if (analysis?.NeedsMoreInfo == true
            && currentRound < MAX_ROUNDS
            && analysis.Questions?.Count >= MIN_QUESTIONS)
        {
            // 还需要更多信息，发送追问
            _roundCounter[pipelineId] = currentRound + 1;

            await sendSse(SseEventType.Question, new
            {
                questions = analysis.Questions,
                understanding = analysis.Understanding,
                round = currentRound + 1,
                message = "为了更好地理解您的需求，我还有几个问题："
            });
        }
        else
        {
            // 信息足够，生成需求文档
            _roundCounter.Remove(pipelineId);

            var doc = analysis?.RequirementDocument
                ?? GenerateDefaultDocument(analysis);

            // 保存文档
            await _pipelineService.SaveMessage(
                pipelineId, 1, "assistant",
                doc, "document");

            await sendSse(SseEventType.StageOutput, new
            {
                stage = 1,
                type = "requirement_document",
                content = doc,
                domainModel = analysis?.ProposedDomainModel,
                strategies = analysis?.Strategies,
                userStories = analysis?.UserStories,
                risks = analysis?.Risks
            });

            await sendSse(SseEventType.NeedConfirm, new
            {
                stage = 1,
                message = "需求分析已完成，请确认是否满足您的需求。如有修改意见可以提出。"
            });

            // 更新状态为等待确认
            pipeline.F_Status = "waitingConfirm";
            await using var db = _pipelineService
                .GetType()
                .GetField("_db",
                    System.Reflection.BindingFlags.NonPublic
                    | System.Reflection.BindingFlags.Instance)
                ?.GetValue(_pipelineService) as ISqlSugarClient;
            // 通过公共方法更新状态

        }

    private async Task<string> GetIndustryContext(
        PromptTemplateService promptService, string? tenantId)
    {
        if (string.IsNullOrEmpty(tenantId)) return "";

        // 从租户行业知识表读取
        var db = promptService.GetType()
            .GetField("_db",
                System.Reflection.BindingFlags.NonPublic
                | System.Reflection.BindingFlags.Instance)
            ?.GetValue(promptService) as ISqlSugarClient;

        if (db == null) return "";

        var industry = await db.Queryable<dynamic>()
            .AS("BASE_TENANT_INDUSTRY")
            .Where("F_TenantId = @tid AND F_DeleteMark = 0",
                new { tid = tenantId })
            .FirstAsync();

        if (industry == null) return "";

        return $"行业背景：\n用户所在行业：{industry.F_IndustryName}\n" +
               $"行业描述：{industry.F_Description}\n" +
               $"关键业务场景：{industry.F_KeyScenarios}";

    private async Task<string> GetGlossaryContext(
        PromptTemplateService promptService, string? tenantId)
    {
        if (string.IsNullOrEmpty(tenantId)) return "";

        var db = promptService.GetType()
            .GetField("_db",
                System.Reflection.BindingFlags.NonPublic
                | System.Reflection.BindingFlags.Instance)
            ?.GetValue(promptService) as ISqlSugarClient;

        if (db == null) return "";

        var terms = await db.Queryable<dynamic>()
            .AS("BASE_TENANT_GLOSSARY")
            .Where("F_TenantId = @tid AND F_DeleteMark = 0",
                new { tid = tenantId })
            .ToListAsync();

        if (terms == null || terms.Count == 0) return "";

        var lines = terms.Select(t =>
            $"- {t.F_Term}: {t.F_Definition}");
        return "业务术语表：\n" + string.Join("\n", lines);
    }

    }

    /// <summary>
    /// 从 AI 响应中提取 JSON（处理 ```json 包裹的情况）
    /// </summary>
    private static string ExtractJson(string text)
    {
        // 尝试直接解析
        if (text.TrimStart().StartsWith("{"))
            return text;

        // 尝试提取 ```json ... ``` 块
        var match = System.Text.RegularExpressions.Regex.Match(
            text, @"```(?:json)?\s*([\s\S]*?)\s*```");
        if (match.Success)
            return match.Groups[1].Value;

        // 尝试提取第一个 { 到最后一个 }
        var start = text.IndexOf('{');
        var end = text.LastIndexOf('}');
        if (start >= 0 && end > start)
            return text[start..(end + 1)];

        return text;
    }

    #endregion

```typescript
/**
 * 创建 SSE 连接（发送消息并流式接收）
 */
export function createStreamConnection(
  pipelineId: number,
  message: string,
  handlers: {
    onThinking?: (data: any) => void
    onChunk?: (data: { content: string; stage: number }) => void
    onStageOutput?: (data: any) => void
    onNeedConfirm?: (data: any) => void
    onQuestion?: (data: any) => void
    onProgress?: (data: any) => void
    onCompleted?: (data: any) => void
    onError?: (data: any) => void
    onDone?: () => void
  }
): EventSource {
  const baseUrl = import.meta.env.VITE_API_URL || ''
  const token = localStorage.getItem('token') || ''
  const encodedMsg = encodeURIComponent(message)

  // EventSource 不支持自定义 header，用 query param 传 token
  const url = `${baseUrl}/api/ai/pipeline/${pipelineId}/stream?message=${encodedMsg}&access_token=${token}`

  const es = new EventSource(url)

  es.addEventListener('thinking', (e) => {
    handlers.onThinking?.(JSON.parse(e.data))
  })

  es.addEventListener('chunk', (e) => {
    handlers.onChunk?.(JSON.parse(e.data))
  })

  es.addEventListener('stage_output', (e) => {
    handlers.onStageOutput?.(JSON.parse(e.data))
  })

  es.addEventListener('need_confirm', (e) => {
    handlers.onNeedConfirm?.(JSON.parse(e.data))
  })

  es.addEventListener('question', (e) => {
    handlers.onQuestion?.(JSON.parse(e.data))
  })

  es.addEventListener('progress', (e) => {
    handlers.onProgress?.(JSON.parse(e.data))
  })

  es.addEventListener('completed', (e) => {
    handlers.onCompleted?.(JSON.parse(e.data))
    es.close()
    handlers.onDone?.()
  })

  es.addEventListener('error', (e) => {
    if (es.readyState === EventSource.CLOSED) {
      handlers.onDone?.()
    } else {
      handlers.onError?.(e.data ? JSON.parse(e.data) : { message: '连接异常' })
      es.close()
    }
  })

  return es
}
```

```vue
import { computed } from 'vue'
import type { StageInfo, StageNumber } from '../types/pipeline'

const props = defineProps<{
  currentStage: StageNumber
  status: string
}>()

defineEmits<{
  (e: 'stage-click', stage: number): void
}>()

const STAGE_DEFINITIONS: Omit<StageInfo, 'status'>[] = [
  { number: 1, name: '需求分析', icon: 'search', description: '理解业务需求' },
  { number: 2, name: '架构设计', icon: 'block', description: '设计系统架构' },

  { number: 5, name: '交付验收', icon: 'check-circle', description: '源码+部署文档' },
]

const stages = computed<StageInfo[]>(() => {
  return STAGE_DEFINITIONS.map(def => ({
    ...def,
    status: getStatus(def.number)
  }))
})

function getStatus(num: number): StageInfo['status'] {
  if (num < props.currentStage) return 'completed'
  if (num === props.currentStage) {
    return props.status === 'waitingConfirm'
      ? 'waiting_confirm'
      : 'active'
  }
  return 'pending'
}

function getStageClass(stage: StageInfo) {
  return `stage-${stage.status}`
}

function getConnectorClass(idx: number) {
  const prev = stages.value[idx - 1]
  return prev.status === 'completed' ? 'done' : 'pending'
}
```

### 4. AI 对话面板组件（SSE 流式）

```vue

<template>
  <div class="ai-chat-panel">
    <!-- 消息列表 -->
    <div ref="messagesContainer" class="messages-area">
      <div
        v-for="msg in messages"
        :key="msg.id || msg._key"

<script setup lang="ts">
import { ref, computed, watch, nextTick } from 'vue'
import type { PipelineMessage } from '../types/pipeline'
import { createStreamConnection } from '../api/pipeline'

const props = defineProps<{
  pipelineId: number
  currentStage: number
  initialMessages?: PipelineMessage[]
  showConfirmBar?: boolean
}>()

const emit = defineEmits<{
  (e: 'confirm', approved: boolean, feedback?: string): void
  (e: 'stage-output', data: any): void
  (e: 'completed', data: any): void
  (e: 'questions', questions: string[]): void
}>()

interface ChatMessage extends Partial<PipelineMessage> {
  _key?: string
  _streaming?: boolean
  _expanded?: boolean
}

const messages = ref<ChatMessage[]>([])
const inputText = ref('')
const isStreaming = ref(false)
const rejectFeedback = ref('')
const pendingQuestions = ref<string[]>([])
const messagesContainer = ref<HTMLElement>()

let currentEventSource: EventSource | null = null
let streamingMsg: ChatMessage | null = null

// 初始化消息
if (props.initialMessages?.length) {
  messages.value = props.initialMessages.map(m => ({
    ...m,
    _expanded: false
  }))
}

function sendMessage() {
  const text = inputText.value.trim()
  if (!text || isStreaming.value) return

  // 添加用户消息到列表
  messages.value.push({
    _key: `user-${Date.now()}`,
    role: 'user',
    content: text,
    contentType: 'text',
    createTime: new Date().toISOString()
  })

  inputText.value = ''
  pendingQuestions.value = []
  scrollToBottom()

  // 创建 AI 流式消息占位
  streamingMsg = {
    _key: `assistant-${Date.now()}`,
    role: 'assistant',
    content: '',
    contentType: 'text',
    _streaming: true,
    createTime: new Date().toISOString()
  }
  messages.value.push(streamingMsg)
  isStreaming.value = true
  scrollToBottom()

  // 建立 SSE 连接
  currentEventSource = createStreamConnection(
    props.pipelineId,
    text,
    {
      onThinking(data) {
        if (streamingMsg) {
          streamingMsg.content = '💭 正在思考...'
        }
      },

      onChunk(data) {
        if (streamingMsg) {
          // 移除初始思考提示
          if (streamingMsg.content === '💭 正在思考...') {
            streamingMsg.content = ''
          }
          streamingMsg.content += data.content
          scrollToBottom()
        }
      },

      onStageOutput(data) {
        // 阶段产出：添加为文档消息
        messages.value.push({
          _key: `output-${Date.now()}`,
          role: 'assistant',
          content: typeof data.content === 'string'
            ? data.content
            : JSON.stringify(data.content, null, 2),
          contentType: data.type === 'requirement_document'
            || data.type === 'architecture_document'
            || data.type === 'design_document'
            ? 'document' : 'json',
          createTime: new Date().toISOString(),
          _expanded: false
        })
        emit('stage-output', data)
        scrollToBottom()
      },

      onNeedConfirm(data) {
        // 结束流式状态
        if (streamingMsg) {
          streamingMsg._streaming = false
          streamingMsg = null
        }
        isStreaming.value = false
        scrollToBottom()
      },

      onQuestion(data) {
        if (streamingMsg) {
          streamingMsg._streaming = false
          streamingMsg = null
        }
        isStreaming.value = false
        pendingQuestions.value = data.questions || []
        emit('questions', data.questions || [])
        scrollToBottom()
      },

      onProgress(data) {
        if (streamingMsg) {
          streamingMsg.content += `\n\n⏳ ${data.message}`
        }
      },

      onCompleted(data) {
        if (streamingMsg) {
          streamingMsg._streaming = false
          streamingMsg = null
        }
        isStreaming.value = false
        emit('completed', data)
        scrollToBottom()
      },

      onError(data) {
        if (streamingMsg) {
          streamingMsg.content += `\n\n❌ 错误: ${data.message}`
          streamingMsg._streaming = false
          streamingMsg = null
        }
        isStreaming.value = false
      },

      onDone() {
        isStreaming.value = false
      }
    }
  )
}

function scrollToBottom() {
  nextTick(() => {
    if (messagesContainer.value) {
      messagesContainer.value.scrollTop = messagesContainer.value.scrollHeight
    }
  })
}

/** 简单的 Markdown 渲染（生产环境替换为 markdown-it） */
function renderMarkdown(text: string): string {
  if (!text) return ''
  return text
    .replace(/^### (.+)$/gm, '<h3>$1</h3>')
    .replace(/^## (.+)$/gm, '<h2>$1</h2>')
    .replace(/^# (.+)$/gm, '<h1>$1</h1>')
    .replace(/\*\*(.+?)\*\*/g, '<strong>$1</strong>')
    .replace(/\*(.+?)\*/g, '<em>$1</em>')
    .replace(/^- (.+)$/gm, '<li>$1</li>')
    .replace(/\n/g, '<br/>')
}

function formatJson(text: string): string {
  try {
    return JSON.stringify(JSON.parse(text), null, 2)
  } catch {
    return text
  }
}

function formatTime(time?: string): string {
  if (!time) return ''
  return new Date(time).toLocaleTimeString('zh-CN', {
    hour: '2-digit',
    minute: '2-digit'
  })
}
```
