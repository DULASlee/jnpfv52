// ============================================================
// 文件: backend/application/JNPF.Application/Studio/Service/StudioMenuService.cs
// 描述: Studio 菜单服务 - 核心权限过滤逻辑
// ============================================================

using JNPF.Application.Studio.Models;
using JNPF.Dependency;
using JNPF.DynamicApiController;
using Microsoft.AspNetCore.Authorization;
using Microsoft.AspNetCore.Mvc;
using SqlSugar;
using System.Text.Json;

namespace JNPF.Application.Studio.Service;

/// <summary>
/// Studio 菜单服务
/// </summary>
[ApiDescriptionSettings(Tag = "Studio菜单")]
[Authorize]
public class StudioMenuService : IDynamicApiController
{
    private readonly ISqlSugarClient _db;
    private readonly IUser _user;

    public StudioMenuService(ISqlSugarClient db, IUser user)
    {
        _db = db;
        _user = user;
    }

    /// <summary>
    /// 获取当前用户可见的菜单树
    /// 核心逻辑：根据用户角色过滤菜单，附带红点数量
    /// </summary>
    [HttpGet("api/menu/user-menus")]
    public async Task<List<StudioMenuDto>> GetUserMenus()
    {
        // 1. 获取用户角色列表
        var userRoles = await GetUserRoles();

        // 2. 获取所有未删除、已启用的菜单
        var allMenus = await _db.Queryable<StudioMenuEntity>()
            .Where(m => !m.F_DeleteMark && m.F_Enabled)
            .OrderBy(m => m.F_Sort)
            .ToListAsync();

        // 3. 过滤：用户角色命中 F_RequiredRoles 的菜单才可见
        var visibleMenus = allMenus.Where(menu => IsMenuVisible(menu, userRoles)).ToList();

        // 4. 获取当前用户的红点数据
        var badges = await _db.Queryable<MenuBadgeEntity>()
            .Where(b => b.F_UserId == _user.UserId && b.F_Count > 0)
            .ToListAsync();

        // 5. 构建树结构
        var menuTree = BuildMenuTree(visibleMenus, 0, badges);

        return menuTree;
    }

    /// <summary>
    /// 标记菜单已读（清除红点）
    /// </summary>
    [HttpPost("api/menu/badge/read")]
    public async Task MarkBadgeRead([FromBody] MarkBadgeReadInput input)
    {
        var badge = await _db.Queryable<MenuBadgeEntity>()
            .Where(b => b.F_MenuId == input.MenuId
                     && b.F_UserId == _user.UserId
                     && b.F_TenantId == _user.TenantId)
            .FirstAsync();

        if (badge != null)
        {
            badge.F_Count = 0;
            badge.F_ModifyTime = DateTime.Now;
            await _db.Updateable(badge).ExecuteCommandAsync();
        }
    }

    /// <summary>
    /// 增加红点计数（供流水线完成时调用）
    /// </summary>
    public async Task IncrementBadge(long menuId, long userId, string tenantId, int increment = 1)
    {
        var existing = await _db.Queryable<MenuBadgeEntity>()
            .Where(b => b.F_MenuId == menuId && b.F_UserId == userId && b.F_TenantId == tenantId)
            .FirstAsync();

        if (existing != null)
        {
            existing.F_Count += increment;
            existing.F_ModifyTime = DateTime.Now;
            await _db.Updateable(existing).ExecuteCommandAsync();
        }
        else
        {
            await _db.Insertable(new MenuBadgeEntity
            {
                F_Id = SnowFlakeSingle.Instance.NextId(),
                F_MenuId = menuId,
                F_UserId = userId,
                F_TenantId = tenantId,
                F_Count = increment,
                F_CreatorTime = DateTime.Now
            }).ExecuteCommandAsync();
        }
    }

    #region 私有方法

    /// <summary>
    /// 判断菜单对当前用户是否可见
    /// </summary>
    private bool IsMenuVisible(StudioMenuEntity menu, List<string> userRoles)
    {
        // 全员可见菜单
        if (menu.F_IsPublic) return true;

        // 解析 RequiredRoles JSON
        if (string.IsNullOrWhiteSpace(menu.F_RequiredRoles)) return true;

        try
        {
            var requiredRoles = JsonSerializer.Deserialize<List<string>>(menu.F_RequiredRoles)
                ?? new List<string>();

            // 没有角色限制的菜单（容器菜单），对已登录用户可见
            if (requiredRoles.Count == 0) return true;

            // 用户的角色列表与菜单要求的角色有交集即可见
            return requiredRoles.Any(r => userRoles.Contains(r, StringComparer.OrdinalIgnoreCase));
        }
        catch
        {
            return false;
        }
    }

    /// <summary>
    /// 构建菜单树（递归）
    /// </summary>
    private List<StudioMenuDto> BuildMenuTree(
        List<StudioMenuEntity> allMenus,
        long parentId,
        List<MenuBadgeEntity> badges)
    {
        return allMenus
            .Where(m => m.F_ParentId == parentId)
            .Select(m => new StudioMenuDto
            {
                Id = m.F_Id,
                ParentId = m.F_ParentId,
                Name = m.F_Name,
                Icon = m.F_Icon,
                Url = m.F_Url,
                Sort = m.F_Sort,
                Comment = m.F_Comment,
                DataScope = m.F_DataScope,
                ExpandPhase = m.F_ExpandPhase,
                BadgeCount = badges
                    .Where(b => b.F_MenuId == m.F_Id)
                    .Sum(b => b.F_Count),
                Children = BuildMenuTree(allMenus, m.F_Id, badges)
            })
            .ToList();
    }

    /// <summary>
    /// 获取用户角色编码列表
    /// </summary>
    private async Task<List<string>> GetUserRoles()
    {
        // 从 JNPF 已有的用户角色关联表获取
        // 具体表名和字段需与现有 BASE_USER_ROLE 对齐
        var roles = await _db.Ado.SqlQueryAsync<string>(
            @"SELECT r.F_EnCode
              FROM BASE_USER_ROLE ur
              INNER JOIN BASE_ROLE r ON ur.F_RoleId = r.F_Id
              WHERE ur.F_UserId = @userId AND r.F_DeleteMark = 0",
            new { userId = _user.UserId });

        return roles;
    }

    #endregion
}

// ============================================================
// 文件: backend/application/JNPF.Application/Studio/Middleware/FounderGuardMiddleware.cs
// 描述: 创始人 TOTP 二次认证中间件
// ============================================================

using JNPF.Dependency;
using Microsoft.AspNetCore.Http;
using System.Net;

namespace JNPF.Application.Studio.Middleware;

/// <summary>
/// 创始人门禁中间件
/// 拦截所有 /api/founder/* 和 /studio/foundry/* 路由
/// 非 founder 角色 → 403
/// founder 未通过 TOTP → 401
/// </summary>
public class FounderGuardMiddleware
{
    private readonly RequestDelegate _next;
    private const string FOUNDER_ROLE = "founder";

    public FounderGuardMiddleware(RequestDelegate next)
    {
        _next = next;
    }

    public async Task InvokeAsync(HttpContext context)
    {
        var path = context.Request.Path.Value?.ToLower() ?? "";

        // 只拦截 foundry 相关路由
        if (!path.Contains("/foundry/") && !path.Contains("/founder/"))
        {
            await _next(context);
            return;
        }

        // 允许 TOTP 验证接口本身通过
        if (path.Contains("/founder/auth/verify") || path.Contains("/founder/totp-verify"))
        {
            await _next(context);
            return;
        }

        // 获取当前用户（依赖 JNPF 已有的认证中间件）
        var user = context.RequestServices.GetService<IUser>();
        if (user == null || user.UserId == 0)
        {
            context.Response.StatusCode = (int)HttpStatusCode.Unauthorized;
            await context.Response.WriteAsJsonAsync(new
            {
                code = 401,
                message = "未登录"
            });
            return;
        }

        // 检查是否为 founder 角色
        var isFounder = await CheckFounderRole(context, user.UserId);
        if (!isFounder)
        {
            context.Response.StatusCode = (int)HttpStatusCode.Forbidden;
            await context.Response.WriteAsJsonAsync(new
            {
                code = 403,
                message = "需要创始人权限"
            });
            return;
        }

        // 检查 TOTP 是否已验证
        var totpVerified = context.Session.GetString($"totp_verified_{user.UserId}");
        if (totpVerified != "true")
        {
            context.Response.StatusCode = (int)HttpStatusCode.Unauthorized;
            await context.Response.WriteAsJsonAsync(new
            {
                code = 401,
                message = "需要TOTP二次验证",
                needTotp = true
            });
            return;
        }

        // 记录审计日志
        await LogFounderAction(context, user.UserId, user.UserName);

        await _next(context);
    }

    private async Task<bool> CheckFounderRole(HttpContext context, long userId)
    {
        var db = context.RequestServices.GetRequiredService<ISqlSugarClient>();
        var count = await db.Ado.SqlQuerySingleAsync<int>(
            @"SELECT COUNT(1)
              FROM BASE_USER_ROLE ur
              INNER JOIN BASE_ROLE r ON ur.F_RoleId = r.F_Id
              WHERE ur.F_UserId = @userId AND r.F_EnCode = @role AND r.F_DeleteMark = 0",
            new { userId, role = FOUNDER_ROLE });
        return count > 0;
    }

    private async Task LogFounderAction(HttpContext context, long userId, string? userName)
    {
        try
        {
            var db = context.RequestServices.GetRequiredService<ISqlSugarClient>();
            await db.Ado.ExecuteCommandAsync(
                @"INSERT INTO BASE_FOUNDER_AUTH_LOG
                  (F_Id, F_UserId, F_UserName, F_Action, F_Detail, F_IpAddress, F_Status, F_CreatorTime)
                  VALUES (@id, @userId, @userName, @action, @detail, @ip, @status, @time)",
                new
                {
                    id = SnowFlakeSingle.Instance.NextId(),
                    userId,
                    userName = userName ?? "",
                    action = "api_access",
                    detail = context.Request.Path,
                    ip = context.Connection.RemoteIpAddress?.ToString(),
                    status = "success",
                    time = DateTime.Now
                });
        }
        catch
        {
            // 审计日志写入失败不应阻塞请求
        }
    }
}

/// <summary>

public static class FounderGuardMiddlewareExtensions
{

    {
        return builder.UseMiddleware<FounderGuardMiddleware>();
    }

  const loading = ref(false)
  const loaded = ref(false)

  /** 一级菜单列表 */
  const topMenus = computed(() =>
    menus.value
      .filter(m => m.parentId === 0)
      .sort((a, b) => a.sort - b.sort)
  )

  /** 获取某个一级菜单下的子菜单树 */
  function getChildren(parentId: number): StudioMenuItem[] {
    const parent = menus.value.find(m => m.id === parentId)
    return parent?.children ?? []
  }

  /** 所有红点总数（用于顶栏显示） */
  const totalBadgeCount = computed(() => {
    let count = 0
    function walk(items: StudioMenuItem[]) {
      for (const item of items) {
        count += item.badgeCount
        if (item.children?.length) walk(item.children)
      }
    }
    walk(menus.value)
    return count
  })

  /** 加载菜单 */
  async function loadMenus() {
    if (loaded.value) return
    loading.value = true
    try {
      const data = await getUserMenus()
      menus.value = data ?? []
      loaded.value = true
    } finally {
      loading.value = false
    }
  }

  /** 刷新菜单（重新拉取） */
  async function refreshMenus() {
    loaded.value = false
    await loadMenus()
  }

  /** 设置某个菜单的红点数量 */
  function setBadge(menuId: number, count: number) {
    function walk(items: StudioMenuItem[]) {
      for (const item of items) {
        if (item.id === menuId) {
          item.badgeCount = count
          return true
        }
        if (item.children?.length && walk(item.children)) return true
      }
      return false
    }
    walk(menus.value)
  }

  return {
    menus,
    loading,
    loaded,
    topMenus,
    totalBadgeCount,
    getChildren,
    loadMenus,
    refreshMenus,
    setBadge,
  }
})

import type { Router } from 'vue-router'
import { useUserStore } from '@/store/modules/user'
import { useStudioMenuStore } from './store/studio-menu'
import NProgress from 'nprogress'

const TOTP_VERIFY_ROUTE = '/studio/foundry/totp-verify'
const LOGIN_ROUTE = '/login'
const FORBIDDEN_ROUTE = '/403'

/**
 * 注册 Studio 路由守卫
 */
export function setupStudioPermissionGuards(router: Router) {
  router.beforeEach(async (to, from, next) => {
    NProgress.start()

    const userStore = useUserStore()

    // 1. 未登录 → 跳登录页
    if (!userStore.token) {
      if (to.path !== LOGIN_ROUTE) {
        return next({ path: LOGIN_ROUTE, query: { redirect: to.fullPath } })
      }
      return next()
    }

    // 2. 首次进入 Studio → 加载菜单
    const menuStore = useStudioMenuStore()
    if (!menuStore.loaded) {
      await menuStore.loadMenus()
    }

    // 3. 检查菜单权限（后端已过滤，前端检查路由是否在菜单中）
    if (to.path.startsWith('/studio/') && to.path !== TOTP_VERIFY_ROUTE) {
      const isMenuRoute = isRouteInMenus(to.path, menuStore.menus)

      // 公共路由放行（如 totp-verify）
      if (to.meta.public) {
        return next()
      }

      // 不在菜单中的 studio 路由 → 403
      if (!isMenuRoute && to.meta.requiresMenu !== false) {
        return next(FORBIDDEN_ROUTE)
      }
    }

    // 4. TOTP 门禁（自博弈训练引擎）
    if (to.path.startsWith('/studio/foundry/') && to.path !== TOTP_VERIFY_ROUTE) {
      const totpVerified = sessionStorage.getItem('founder_totp_verified')
      if (totpVerified !== 'true') {
        return next({
          path: TOTP_VERIFY_ROUTE,
          query: { redirect: to.fullPath },
        })
      }
    }

    next()
  })

  router.afterEach(() => {
    NProgress.done()
  })
}

/**
 * 检查路由路径是否存在于菜单树中
 */
function isRouteInMenus(path: string, menus: any[]): boolean {
  for (const menu of menus) {
    if (menu.url === path) return true
    if (menu.children?.length && isRouteInMenus(path, menu.children)) return true
  }
  return false
}

<script setup lang="ts">
import { ref, nextTick } from 'vue'
import { useRouter, useRoute } from 'vue-router'
import { verifyTotp } from '../api/menu'

const router = useRouter()
const route = useRoute()

const inputRefs = ref<HTMLInputElement[]>([])
const codeDigits = ref<string[]>([])
const hasError = ref(false)
const errorMessage = ref('')
const verifying = ref(false)

function onInput(idx: number, event: Event) {
  const value = (event.target as HTMLInputElement).value
  if (!/^\d$/.test(value)) {
    codeDigits.value[idx] = ''
    return
  }

  codeDigits.value[idx] = value
  hasError.value = false
  errorMessage.value = ''

  // 自动跳到下一个输入框
  if (idx < 5) {
    nextTick(() => inputRefs.value[idx + 1]?.focus())
  }

  // 全部输入完自动验证
  if (codeDigits.value.join('').length === 6) {
    handleVerify()
  }
}

function onKeydown(idx: number, event: KeyboardEvent) {
  if (event.key === 'Backspace' && !codeDigits.value[idx] && idx > 0) {
    codeDigits.value[idx - 1] = ''
    nextTick(() => inputRefs.value[idx - 1]?.focus())
  }
}

function onPaste(event: ClipboardEvent) {
  const text = event.clipboardData?.getData('text')?.trim() ?? ''
  if (/^\d{6}$/.test(text)) {
    event.preventDefault()
    for (let i = 0; i < 6; i++) {
      codeDigits.value[i] = text[i]
    }
    handleVerify()
  }
}

async function handleVerify() {
  const code = codeDigits.value.join('')
  if (code.length !== 6) return

  verifying.value = true
  errorMessage.value = ''

  try {
    const res = await verifyTotp({ code })
    if (res.success) {
      // 标记 TOTP 已验证
      sessionStorage.setItem('founder_totp_verified', 'true')

      // 跳转到目标页或默认跳仪表盘
      const redirect = (route.query.redirect as string) || '/studio/foundry/dashboard'
      router.push(redirect)
    } else {
      hasError.value = true
      errorMessage.value = res.message || '验证码无效，请重试'
      // 清空输入
      codeDigits.value = []
      nextTick(() => inputRefs.value[0]?.focus())
    }
  } catch (e) {
    hasError.value = true
    errorMessage.value = '验证失败，请稍后重试'
  } finally {
    verifying.value = false
  }
}
</script>
