#### 2.3.2 SqlSugar AOP 实现

```csharp
// TenantAwareSqlSugarClient.cs
public class TenantAwareSqlSugarClient
{
    private readonly string _tenantId;
    private readonly string _connectionString;

    public TenantAwareSqlSugarClient(string tenantId, string connectionString)
    {
        _tenantId = tenantId;
        _connectionString = connectionString;
    }

    public ISqlSugarClient GetClient()
    {
        var db = new SqlSugarClient(new ConnectionConfig
        {
            ConnectionString = _connectionString,
            DbType = DbType.SqlServer,
            IsAutoCloseConnection = true,
            InitKeyType = InitKeyType.Attribute
        });

        // 关键 AOP:动态注入 Schema 前缀
        db.Aop.DataExecuting = (oldValue, entityInfo) =>
        {
            // 拦截所有读写操作
            if (entityInfo.OperationType == OperationType.Insert ||
                entityInfo.OperationType == OperationType.Update ||
                entityInfo.OperationType == OperationType.Delete ||
                entityInfo.OperationType == OperationType.Query ||
                entityInfo.OperationType == OperationType.ReallyAsync)
            {
                // 动态切换表名 = Schema + 表名
                var originalTable = entityInfo.EntityInfo.DbTableName;
                if (!originalTable.Contains("."))
                {
                    entityInfo.EntityInfo.DbTableName = 
                        $"tenant_{_tenantId}.{originalTable}";
                }
            }
        };

        return db;
    }
}
```

#### 3.3.2 Channel 异步队列实现

```csharp
// CdcSyncService.cs
public class CdcSyncService : BackgroundService
{
    private readonly Channel<SyncTask> _queue;
    private readonly ILogger<CdcSyncService> _logger;

    public CdcSyncService()
    {
        // 有界队列,容量 10000,满了则阻塞生产者(背压)
        _queue = Channel.CreateBounded<SyncTask>(new BoundedChannelOptions(10000)
        {
            FullMode = BoundedChannelFullMode.Wait,
            SingleReader = false,
            SingleWriter = false
        });
    }

    protected override async Task ExecuteAsync(CancellationToken stoppingToken)
    {
        // 启动 3 个消费者 worker(并行)
        var consumers = new List<Task>
        {
            ConsumeQueueAsync(stoppingToken, "Worker-1"),
            ConsumeQueueAsync(stoppingToken, "Worker-2"),
            ConsumeQueueAsync(stoppingToken, "Worker-3")
        };
        
        // 启动生产者(CDC 轮询)
        var producer = Task.Run(async () => await ProduceLoopAsync(stoppingToken));
        
        await Task.WhenAll(consumers.Append(producer));
    }

    private async Task ProduceLoopAsync(CancellationToken ct)
    {
        while (!ct.IsCancellationRequested)
        {
            // 1. 轮询每张表的 CDC 变更
            foreach (var (tenantId, table) in _monitoredTables)
            {
                var changes = await _cdcReader.ReadChanges(table.Name, table.LastLsn);
                if (changes.Any())
                {
                    foreach (var change in changes)
                    {
                        var task = new SyncTask
                        {
                            TenantId = tenantId,
                            TableName = table.Name,
                            Change = change,
                            EnqueueTime = DateTime.UtcNow
                        };
                        await _queue.Writer.WriteAsync(task, ct);
                    }
                    table.LastLsn = changes.Max(c => c.Lsn);
                }
            }
            await Task.Delay(1000, ct);  // 1 秒轮询
        }
    }

    private async Task ConsumeQueueAsync(CancellationToken ct, string workerName)
    {
        await foreach (var task in _queue.Reader.ReadAllAsync(ct))
        {
            try
            {
                // 2. 租户路由
                var client = _tenantClientFactory.GetClient(task.TenantId);
                
                // 3. Upsert 到对应 Schema
                await UpsertChangeAsync(client, task);
                
                // 4. 记录延迟
                var latency = DateTime.UtcNow - task.EnqueueTime;
                _metrics.RecordSyncLatency(task.TenantId, task.TableName, latency);
            }
            catch (Exception ex)
            {
                // 5. 失败处理:重试 3 次 + 死信队列
                await _retryPolicy.ExecuteAsync(() => UpsertChangeAsync(client, task));
                _logger.LogError(ex, $"Worker {workerName} 同步失败");
            }
        }
    }
}
```

### 3.5 失败回退方案

**触发条件**: A2.2 数据丢失 OR A2.3 P95 > 5 分钟 OR A2.5 跨租户串数据

**回退方案 A**: 每租户独立 worker 进程(牺牲并发度保稳定性)
- 20 租户 × 3 worker = 60 进程,资源开销大
- 优点: 故障隔离,易调试
- 缺点: 进程数膨胀,运维复杂

**回退方案 B**: 改用 SQL Server 事务复制(更重量级)
- 主从复制 + 自动应用变更
- 优点: SQL Server 原生,稳定
- 缺点: 配置复杂,延迟高(>5s)

**回退方案 C**: 改用应用层时间戳增量(去掉 CDC)
- 简单,稳定,但实时性差(分钟级)
- 优点: 零基础设施依赖
- 缺点: 延迟 5-15 分钟,可能不满足 SLA

#### 4.3.2 24 小时持续写入压测

```python
# continuous_write_load_test.py
import asyncio
import time
import random
from datetime import datetime, timedelta
import aiohttp

# 配置
TENANTS = [f"load_test_{i:03d}" for i in range(20)]
TARGET_DAILY_ROWS = 60_000_000  # 6000 万行/日
TARGET_TPS = TARGET_DAILY_ROWS / 86400  # ~694 行/秒
DURATION_HOURS = 24

async def insert_batch(session, tenant_id, batch_size=100):
    """异步插入一批数据"""
    rows = []
    for _ in range(batch_size):
        rows.append({
            "id": str(uuid.uuid4()),
            "tenantId": tenant_id,
            "workshopId": f"WS_{random.randint(1, 10)}",
            "productId": f"PROD_{random.randint(1, 100)}",
            "orderStatus": "completed" if random.random() < 0.8 else "pending",
            "planQty": random.randint(100, 1000),
            "completedQty": random.randint(80, 1000),
            "passQty": random.randint(70, 1000),
            "completeTime": (datetime.now() - timedelta(days=random.randint(0, 30))).isoformat()
        })
    
    async with session.post(
        f"http://localhost:5000/api/data/insert-batch",
        json={"tenantId": tenant_id, "rows": rows}
    ) as resp:
        return await resp.json()

async def continuous_load(duration_seconds):
    """持续负载测试"""
    start_time = time.time()
    total_inserted = 0
    errors = 0
    
    async with aiohttp.ClientSession() as session:
        while time.time() - start_time < duration_seconds:
            batch_start = time.time()
            
            # 并发所有租户 × 每秒 35 个批次(694 TPS / 20 租户)
            tasks = []
            for tenant in TENANTS:
                # 每租户每秒约 35 个批次 × 100 行 = 3500 行
                tasks.append(insert_batch(session, tenant, batch_size=100))
            
            results = await asyncio.gather(*tasks, return_exceptions=True)
            
            for r in results:
                if isinstance(r, Exception):
                    errors += 1
                else:
                    total_inserted += r.get('rowsInserted', 0)
            
            # 控制速率
            elapsed = time.time() - batch_start
            sleep_time = max(0, 1.0 - elapsed)
            await asyncio.sleep(sleep_time)
            
            # 每 60 秒打印进度
            if int(time.time() - start_time) % 60 == 0:
                elapsed_min = (time.time() - start_time) / 60
                rate = total_inserted / (time.time() - start_time)
                print(f"[{elapsed_min:.0f}min] 已插入 {total_inserted:,} 行,速率 {rate:.0f} 行/秒, 错误 {errors}")

# 启动测试
print(f"启动 24 小时持续负载测试")
print(f"目标: 6000 万行/日 = {TARGET_TPS:.0f} 行/秒")
asyncio.run(continuous_load(DURATION_HOURS * 3600))
```

```python
# eval_local_llm.py
import json
import time
import requests
from typing import Dict, List

class LocalLlmEvaluator:
    def __init__(self, model="qwen:7b-chat-v1.5-q4_K_M"):
        self.model = model
        self.ollama_url = "http://localhost:11434/api/generate"
        self.eval_set = self.load_eval_set()
    
    def load_eval_set(self) -> List[Dict]:
        with open("eval_set_50.json", "r", encoding="utf-8") as f:
            return json.load(f)
    
    def call_llm(self, prompt: str) -> tuple[str, float]:
        """调用本地 LLM,返回 (输出, 延迟秒)"""
        start = time.time()
        response = requests.post(self.ollama_url, json={
            "model": self.model,
            "prompt": prompt,
            "stream": False,
            "options": {
                "temperature": 0.1,  # 低随机性,保准确
                "top_p": 0.9,
                "num_predict": 512
            }
        })
        elapsed = time.time() - start
        return response.json()["response"], elapsed
    
    def mql_match(self, predicted: Dict, expected: Dict) -> bool:
        """比较预测 MQL 与标准 MQL"""
        # 简化版: 比较关键字段
        for key in ["metrics", "dimensions", "orderBy", "limit"]:
            if predicted.get(key) != expected.get(key):
                return False
        # timeRange 宽松匹配
        pred_time = predicted.get("timeRange", {}).get("preset") or predicted.get("timeRange", {}).get("start")
        exp_time = expected.get("timeRange", {}).get("preset") or expected.get("timeRange", {}).get("start")
        if pred_time != exp_time:
            return False
        return True
    
    def run(self):
        correct = 0
        total = len(self.eval_set)
        latencies = []
        
        for item in self.eval_set:
            # 构造 Prompt
            prompt = SYSTEM_PROMPT.format(
                metric_dictionary=METRIC_DICT,
                schema_rag_context="",  # POC 阶段无 RAG
                examples="",
                nl_query=item["query"]
            )
            
            # 调用 LLM
            output, latency = self.call_llm(prompt)
            latencies.append(latency)
            
            # 解析输出
            try:
                # 尝试提取 JSON(可能 LLM 输出了多余内容)
                json_start = output.find("{")
                json_end = output.rfind("}") + 1
                predicted_mql = json.loads(output[json_start:json_end])
                
                if self.mql_match(predicted_mql, item["expected_mql"]):
                    correct += 1
                    print(f"✓ Query {item['编号']}: {item['query']}")
                else:
                    print(f"✗ Query {item['编号']}: {item['query']}")
                    print(f"  预期: {json.dumps(item['expected_mql'], ensure_ascii=False)}")
                    print(f"  实际: {json.dumps(predicted_mql, ensure_ascii=False)}")
            except json.JSONDecodeError:
                print(f"✗ Query {item['编号']}: JSON 解析失败 - {output[:100]}")
        
        accuracy = correct / total * 100
        avg_latency = sum(latencies) / len(latencies)
        p95_latency = sorted(latencies)[int(len(latencies) * 0.95)]
        
        print(f"\n=== 评测结果 ===")
        print(f"准确率: {correct}/{total} = {accuracy:.1f}%")
        print(f"平均延迟: {avg_latency:.2f}s")
        print(f"P95 延迟: {p95_latency:.2f}s")
        
        return accuracy, p95_latency

# 运行
```

```python
import time
import requests
from datetime import datetime

def test_p0_latency():
    """P0 实时同步延迟测试"""
    print("=== P0 实时同步延迟测试 ===")
    
    # 1. 在生产库插入一条工单
    test_data = {
        "id": f"TEST_P0_{int(time.time())}",
        "tenantId": "test_001",
        "workshopId": "WS_1",
        "orderStatus": "completed",
        "planQty": 100,
        "completedQty": 100,
        "passQty": 95
    }
    
    # 2. 调用生产库插入 API
    insert_start = time.time()
    response = requests.post(
        "http://localhost:5001/api/production/workorder",
        json=test_data
    )
    insert_time = time.time()
    print(f"生产库插入时间: {datetime.fromtimestamp(insert_time).isoformat()}")
    
    # 3. 轮询分析库,等待数据出现
    poll_start = time.time()
    max_wait = 120  # 最多等 2 分钟
    found = False
    
    while time.time() - poll_start < max_wait:
        check_response = requests.get(
            f"http://localhost:5000/api/analytics/workorder/{test_data['id']}"
        )
        if check_response.status_code == 200:
            elapsed = time.time() - insert_time
            print(f"✅ 同步成功!延迟: {elapsed:.2f} 秒")
            found = True
            break
        time.sleep(0.5)
    
    if not found:
        print(f"❌ 同步超时(>{max_wait}秒)")
        return False
    
    # 4. 记录指标
    return {
        "p0_latency_seconds": elapsed,
        "p0_target": 60,  # < 1 分钟
        "passed": elapsed < 60
    }

result = test_p0_latency()
print(f"结果: {result}")
```

### 6.5 失败回退方案

**触发条件**: A5.1 P95 > 5 分钟 OR A5.5 跨租户串数据

**回退方案 A**: 全部改为日级同步(P2 模式)
- 优点: 简单,稳定
- 缺点: P0 实时性丧失(用户预期降低)
- 业务影响: 早报数据延迟到次日凌晨,班组长无法实时看数据

**回退方案 B**: 减少 P0 表数量(从 5-10 张降到 3-5 张)
- 只同步工单状态 + 报工 + 质检(最关键 3 张)
- 优点: 资源压力降低
- 缺点: 部分数据延迟变大

**决策流程**:
```
POC-5 失败
  ↓
├─ A5.5 失败(跨租户串) → 严重,立即停止 → 检查独立 Schema 路由(可能 POC-1 也有问题)
├─ A5.1 失败(P0 延迟) → 启用回退方案 B(减少 P0 表)
└─ A5.2/A5.3 失败(P1/P2 延迟) → 启用回退方案 A(全降为日级)
```
