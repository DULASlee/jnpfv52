```csharp
    {
        context.Response.ContentType = "application/json";
        var result = new
        {
            status = report.Status.ToString(),
            checks = report.Entries.Select(e => new
            {

        await context.Response.WriteAsync(Newtonsoft.Json.JsonConvert.SerializeObject(result));
    }

{

});
```

```csharp
// DiffLog — 数据变更审计（仅对显式调用 .EnableDiffLogEvent() 的操作生效）
db.Aop.OnDiffLog = (diff) =>
{
    try
    {
        var diffDb = db.CopyNew();
        var userId = TraceContext.TraceId != "unknown"
            ? App.GetService<IUserManager>()?.UserId
            : null;

        _ = diffDb.Insertable(new JNPF.Common.Entity.Sys.SysDiffLogEntity
        {
            Id = Yitter.IdGenerator.YitIdHelper.NextId().ToString(),
            TableName = diff.TableName ?? "Unknown",
            DiffType = diff.DiffType.ToString(),
            BeforeData = diff.BeforeData?.ToJson(),
            AfterData = diff.AfterData?.ToJson(),
            UserId = userId,
            TraceId = TraceContext.TraceId,
            CreateTime = DateTime.Now
        }).ExecuteCommandAsync();
    }
    catch
    {
        // 审计日志写入失败不应影响业务操作
    }
};
```

### 代码变更

**Startup.cs ConfigureServices 方法中（第 94 行 AddControllers 之前）追加：**

```csharp
// API 限流
services.AddRateLimiter(options =>
{
    // 全局固定窗口限流：每秒 200 请求
    options.AddFixedWindowLimiter("fixed", opt =>
    {
        opt.PermitLimit = 200;
        opt.Window = TimeSpan.FromSeconds(1);
        opt.QueueProcessingOrder = System.Threading.RateLimiting.QueueProcessingOrder.OldestFirst;
        opt.QueueLimit = 20;
    });

    // 登录接口限流：每分钟 20 次
    options.AddSlidingWindowLimiter("login", opt =>
    {
        opt.PermitLimit = 20;
        opt.Window = TimeSpan.FromMinutes(1);
        opt.SegmentsPerWindow = 6;
    });

    // 导出接口限流：每分钟 10 次
    options.AddSlidingWindowLimiter("export", opt =>
    {
        opt.PermitLimit = 10;
        opt.Window = TimeSpan.FromMinutes(1);
        opt.SegmentsPerWindow = 6;
    });

    // 限流触发时返回 429
    options.OnRejected = (context, cancellationToken) =>
    {
        context.HttpContext.Response.StatusCode = StatusCodes.Status429TooManyRequests;
        context.HttpContext.Response.ContentType = "application/json";
        var result = Newtonsoft.Json.JsonConvert.SerializeObject(new
        {
            code = 429,
            msg = "请求过于频繁，请稍后再试"
        });
        return context.HttpContext.Response.WriteAsync(result, cancellationToken);
    };
});
```

**Startup.cs Configure 方法中（第 277 行 UseCorsAccessor 之后）追加：**

```csharp
// API 限流中间件（在认证之前，未认证请求也会被限流）

```

### 使用方式（可选，逐步应用到关键端点）

```csharp
// 方式 1：在 Service 类级别应用（该 Service 所有方法受限制流）
[Microsoft.AspNetCore.RateLimiting.EnableRateLimiting("fixed")]
public class UserService : IDynamicApiController { ... }

// 方式 2：在特定方法上应用
[EnableRateLimiting("login")]
public async Task<LoginOutput> Login(LoginInput input) { ... }

// 方式 3：不标记任何特性 — 默认使用全局 "fixed" 限流策略
public class SomeService : IDynamicApiController { ... }
```

### 验证方法

```bash

# 2. 正常调用 API — 确认功能不受影响
```

### 影响面

- 新增限流中间件，默认策略为每秒 200 请求（远高于正常业务量）
- 不标记 `[EnableRateLimiting]` 的端点使用全局默认策略
- 登录/导出等敏感端点可选择性应用更严格的策略
- 现有 API 功能完全不受影响
