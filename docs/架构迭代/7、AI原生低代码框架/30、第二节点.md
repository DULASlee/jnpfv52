#### 2.2.2 租户开通工作流

```csharp
// TenantService.cs
public class TenantService : ITenantService
{
    private readonly ISqlSugarClient _baseDb;  // 主库
    private readonly ITenantClientFactory _clientFactory;
    private readonly ILogger<TenantService> _logger;

    /// <summary>
    /// 开通新租户(原子操作)
    /// </summary>
    public async Task<TenantEntity> ProvisionTenantAsync(ProvisionTenantRequest request)
    {
        // 1. 事务开始(主库)
        var tenant = new TenantEntity
        {
            Id = SnowflakeIdGenerator.NextId(),
            TenantCode = request.TenantCode,
            TenantName = request.TenantName,
            Plan = request.Plan ?? "standard",
            IsolationLevel = "shared",
            DbKey = "AnalyticsDb",  // 第一阶段共享库
            SchemaName = $"tenant_{request.TenantCode}",
            MaxUsers = request.Plan == "enterprise" ? 200 : 50,
            MaxQps = request.Plan == "enterprise" ? 50 : 20,
            MaxStorageGB = request.Plan == "enterprise" ? 500 : 100,
            MonthlyLlmBudget = request.Plan == "enterprise" ? 2000 : 500,
            Status = "active",
            ExpireDate = DateTime.Now.AddYears(1),
            CreateTime = DateTime.Now,
            CreatorId = request.OperatorId
        };

        await _baseDb.Insertable(tenant).ExecuteCommandAsync();

        try
        {
            // 2. 在分析库创建 Schema + 复制模板表
            var analyticsClient = _clientFactory.GetAnalyticsClient("system");
            await analyticsClient.Ado.ExecuteCommandAsync(
                $"CREATE SCHEMA [{tenant.SchemaName}]"
            );

            // 3. 从 dbo 模板复制 20 张表
            var templateSql = GenerateSchemaCloneSql(tenant.SchemaName);
            await analyticsClient.Ado.ExecuteCommandAsync(templateSql);

            // 4. 创建租户专属 SQL 登录名(弹性 RG 分类用)
            var loginName = $"tenant_user_{tenant.TenantCode}";
            var password = GenerateSecurePassword();
            await analyticsClient.Ado.ExecuteCommandAsync($@"
                CREATE LOGIN [{loginName}] WITH PASSWORD = '{password}';
                CREATE USER [{loginName}] FOR LOGIN [{loginName}];
                ALTER USER [{loginName}] WITH DEFAULT_SCHEMA = [{tenant.SchemaName}];
            ");

            // 5. 注册到租户管理表
            await RegisterTenantLoginAsync(tenant.Id, loginName);

            _logger.LogInformation($"租户 {tenant.TenantCode} 开通成功,Schema {tenant.SchemaName}");
            return tenant;
        }
        catch (Exception ex)
        {
            // 6. 回滚:删除主库记录 + 已创建的 Schema
            await _baseDb.Deleteable<TenantEntity>(tenant.Id).ExecuteCommandAsync();
            _logger.LogError(ex, $"租户 {request.TenantCode} 开通失败,已回滚");
            throw;
        }
    }

    private string GenerateSchemaCloneSql(string schemaName)
    {
        var templateTables = new[]
        {
            "DW_WORK_ORDER", "DW_QUALITY_RECORD", "DW_EQUIPMENT",
            "DW_EQUIPMENT_EVENT", "DW_MATERIAL_STOCK", "DW_WORKER",
            "DW_PRODUCTION_REPORT", "DW_SHIFT_RECORD", "DW_BOM",
            "DW_PRODUCT", "DW_WORKSHOP", "DW_MATERIAL",
            "DW_DIM_DATE", "DW_DIM_WORKSHOP", "DW_DIM_PRODUCT",
            "BASE_META_ENTITY", "BASE_META_FIELD",
            "BASE_METRIC_DEFINITION", "BASE_DIMENSION_DEFINITION",
            "BASE_SEMANTIC_KEYWORD_INDEX"
        };

        var sql = $"CREATE SCHEMA [{schemaName}];\n";
        foreach (var table in templateTables)
        {
            sql += $"CREATE TABLE [{schemaName}].[{table}] (LIKE dbo.[template_{table}] INCLUDING ALL);\n";
        }
        return sql;
    }

    private string GenerateSecurePassword()
    {
        // 32 字符强密码
        const string chars = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789!@#$%^&*";
        var random = new byte[32];
        using var rng = RandomNumberGenerator.Create();
        rng.GetBytes(random);
        return new string(random.Select(b => chars[b % chars.Length]).ToArray());
    }
}
```

```csharp
            }
        }

    }
}
```

### 5.2 测试用例

```csharp
// CrossSchemaIsolationTest.cs
[TestClass]
public class CrossSchemaIsolationTest
{
    private TenantAwareSqlSugarClient _clientT001;
    private TenantAwareSqlSugarClient _clientT002;

    [TestInitialize]
    public void Setup()
    {
        _clientT001 = new TenantAwareSqlSugarClient("t001", "...").GetClient();
        _clientT002 = new TenantAwareSqlSugarClient("t002", "...").GetClient();
    }

    [TestMethod]
    public async Task T001_InsertData_ShouldNotVisible_InT002()
    {
        // T001 插入 100 条
        for (int i = 0; i < 100; i++)
        {
            await _clientT001.Insertable(new DwWorkOrderEntity
            {
                Id = Guid.NewGuid().ToString(),
                TenantId = "t001",
                WorkshopId = "WS_1",
                PlanQty = 100
            }).ExecuteCommandAsync();
        }
        
        // T002 查不到(白盒兜底)
        var list = await _clientT002.Queryable<DwWorkOrderEntity>().ToListAsync();
        Assert.AreEqual(0, list.Count);
    }

    [TestMethod]
    public async Task SchemaSwitching_DirectSql_ShouldNotCrossSchema()
    {
        // 直接 SQL 切到 t002 Schema,查询
        var sql = "SELECT COUNT(*) FROM tenant_t001.DW_WORK_ORDER WHERE F_TENANT_ID = 't002'";
        var count = await _clientT002.Ado.GetIntAsync(sql);
        Assert.AreEqual(0, count);
    }

    [TestMethod]
    public async Task SqlInjection_ShouldNotBypassSchemaSwitch()
    {
        // 尝试 SQL 注入
        var malicious = "t001'; DROP TABLE tenant_t002.DW_WORK_ORDER; --";
        try
        {
            var list = await _clientT001.Queryable<DwWorkOrderEntity>()
                .Where(it => it.TenantId == malicious)
```

### 6.2 生产级 AOP 实现

```csharp
// ProductionTenantAwareClient.cs
public class ProductionTenantAwareClient : ITenantAwareClient
{
    private readonly string _tenantId;
    private readonly string _connectionString;
    private readonly ISqlSugarClient _db;

    public ProductionTenantAwareClient(
        string tenantId, 
        string connectionString,
        ITenantConnectionPool pool)
    {
        _tenantId = tenantId;
        _connectionString = connectionString;
        _db = pool.GetClient(connectionString);
        ConfigureAop();
    }

    private void ConfigureAop()
    {
        // 拦截所有数据库操作
        _db.Aop.DataExecuting = (oldValue, entityInfo) =>
        {
            // 1. 拦截:Query / Insert / Update / Delete
            if (!ShouldIntercept(entityInfo.OperationType)) return;

            // 2. 动态切换 Schema(关键)
            SwitchSchema(entityInfo);

            // 3. 强制设置 TenantId(写入场景)
            if (entityInfo.OperationType == OperationType.Insert ||
                entityInfo.OperationType == OperationType.Update)
            {
                ForceTenantId(entityInfo);
            }
        };

        // SQL 执行前:最后兜底检查
        _db.Aop.OnLogExecuting = (sql, pars) =>
        {
            // 防止意外 SQL 绕过 AOP
            ValidateSqlSafety(sql);
        };

        // 慢查询监控
        _db.Aop.OnLogExecuted = (sql, pars) =>
        {
            // 已通过 ISqlSugarClient.Ado 记录
        };
    }

    private void SwitchSchema(EntityInfo entityInfo)
    {
        var originalTable = entityInfo.EntityInfo.DbTableName;
        if (string.IsNullOrEmpty(originalTable)) return;

        // 防止重复切(可能多次调用)
        if (originalTable.Contains(".")) return;

        // 系统表 / 跨租户表不走 Schema 切换
        if (IsSystemTable(originalTable)) return;

        // 关键切换
        entityInfo.EntityInfo.DbTableName = $"tenant_{_tenantId}.{originalTable}";
    }

    private void ForceTenantId(EntityInfo entityInfo)
    {
        // 如果实体是 TenantEntity 派生类,强制设置 TenantId
        if (entityInfo.EntityValue is TenantEntity entity)
        {
            if (string.IsNullOrEmpty(entity.TenantId) || entity.TenantId != _tenantId)
            {
                entity.TenantId = _tenantId;
            }
        }
    }

    private bool ShouldIntercept(OperationType op)
    {
        return op == OperationType.Insert ||
               op == OperationType.Update ||
               op == OperationType.Delete ||
               op == OperationType.Query ||
               op == OperationType.ReallyAsync;
    }

    private bool IsSystemTable(string tableName)
    {
        // 跨租户表:在 dbo.Schema 下,所有租户共享
        var sharedTables = new HashSet<string>
        {
            "BASE_TENANT",
            "BASE_USER",
            "BASE_ROLE",
            "BASE_PERMISSION",
            "DATA_SYNC_TASK",
            "DATA_SYNC_LOG",
            "BASE_AI_QUERY_LOG"
        };
        return sharedTables.Contains(tableName);
    }

    private void ValidateSqlSafety(string sql)
    {
        // 1. 禁止跨 Schema 写操作
        if (Regex.IsMatch(sql, @"(INSERT|UPDATE|DELETE)\s+(INTO\s+)?tenant_(?!{_tenantId})", RegexOptions.IgnoreCase))
        {
            throw new UnauthorizedAccessException($"禁止跨租户写操作: {sql.Substring(0, Math.Min(200, sql.Length))}");
        }

        // 2. 禁止跨 Schema 删除表
        if (Regex.IsMatch(sql, @"DROP\s+(TABLE|SCHEMA)", RegexOptions.IgnoreCase))
        {
            throw new UnauthorizedAccessException("禁止 DROP 操作");
        }
    }

    public ISqlSugarClient Raw => _db;
}
```

```csharp
// QueryClassifier.cs(规则版,Sprint 4 升级为 LLM 版)
public class QueryClassifier
{
    public (QueryComplexity complexity, double confidence) Classify(string nlQuery)
    {
        // 1. 关键词规则(硬编码)
        var simpleScore = 0;
        var standardScore = 0;
        var complexScore = 0;

        // 简单信号
        if (ContainsAny(nlQuery, new[] { "今天", "昨天", "今天早上", "今早", "今晚" }))
            simpleScore += 2;
        if (ContainsAny(nlQuery, new[] { "本月", "上月", "本周", "上周" }) && 
            !ContainsAny(nlQuery, new[] { "为什么", "原因", "怎么" }))
            simpleScore += 1;

        // 中等信号
        if (ContainsAny(nlQuery, new[] { "环比", "同比", "对比", "比较" }))
            standardScore += 2;
        if (ContainsAny(nlQuery, new[] { "各", "分别", "排名" }))
            standardScore += 1;

        // 复杂信号
        if (ContainsAny(nlQuery, new[] { "为什么", "原因", "归因", "怎么改善", "如何提升" }))
            complexScore += 3;
        if (ContainsAny(nlQuery, new[] { "趋势", "预测", "未来" }))
            complexScore += 2;

        // 综合判断
        var maxScore = Math.Max(simpleScore, Math.Max(standardScore, complexScore));
        if (maxScore == 0) return (QueryComplexity.Unknown, 0.5);

        var total = simpleScore + standardScore + complexScore;
        var confidence = (double)maxScore / total;
        if (confidence < 0.8) return (QueryComplexity.Unknown, confidence);

        if (maxScore == simpleScore) return (QueryComplexity.Simple, confidence);
        if (maxScore == standardScore) return (QueryComplexity.Standard, confidence);
        return (QueryComplexity.Complex, confidence);
    }
}
```

## 十、T1.9 Channel 异步队列(D5 上午)

### 10.1 目标

把 Sprint 0 POC 升级为生产级 Channel 异步队列,支持背压、监控、动态扩缩容。

### 10.2 生产级实现

```csharp
// ProductionCdcSyncService.cs
public class ProductionCdcSyncService : BackgroundService
{
    private readonly Channel<SyncTask> _queue;
    private readonly Channel<SyncTask> _deadLetterQueue;
    private readonly IServiceProvider _serviceProvider;
    private readonly ILogger<ProductionCdcSyncService> _logger;
    private readonly IMetricsCollector _metrics;
    private readonly int _workerCount;

    public ProductionCdcSyncService(IConfiguration config, ...)
    {
        // 队列容量根据租户数动态计算
        // 20 租户 × 5000 队列深度 = 10 万
        var capacity = config.GetValue<int>("Sync:QueueCapacity", 100000);
        
        _queue = Channel.CreateBounded<SyncTask>(new BoundedChannelOptions(capacity)
        {
            FullMode = BoundedChannelFullMode.Wait,  // 背压:满了阻塞生产者
            SingleReader = false,
            SingleWriter = false,
            AllowSynchronousContinuations = false
        });
        
        // 死信队列独立
        _deadLetterQueue = Channel.CreateUnbounded<SyncTask>();
        
        // Worker 数量 = CPU 核数 × 2
        _workerCount = Math.Max(4, Environment.ProcessorCount * 2);
    }

    protected override async Task ExecuteAsync(CancellationToken stoppingToken)
    {
        // 启动多个 Worker
        var workers = Enumerable.Range(1, _workerCount)
            .Select(i => ConsumeLoopAsync(stoppingToken, $"Worker-{i}"))
            .ToList();

        // 启动生产者
        var producer = ProduceLoopAsync(stoppingToken);
        
        // 启动 DLQ 处理器
        var dlqHandler = DlqHandlerLoopAsync(stoppingToken);

        await Task.WhenAll(workers.Append(producer).Append(dlqHandler));
    }

    private async Task ProduceLoopAsync(CancellationToken ct)
    {
        // 每租户独立 CDC 监听器
        var tenants = await GetActiveTenantsAsync();
        var tasks = tenants.Select(tenant => 
            TenantCdcListenerAsync(tenant, ct)
        ).ToList();
        
        await Task.WhenAll(tasks);
    }

    private async Task TenantCdcListenerAsync(string tenantId, CancellationToken ct)
    {
        // 每张 P0 表的 CDC 监听
        foreach (var table in P0Tables)
        {
            await TableCdcListenerAsync(tenantId, table, ct);
        }
    }

    private async Task TableCdcListenerAsync(string tenantId, string table, CancellationToken ct)
    {
        var lastLsn = await GetLastProcessedLsnAsync(tenantId, table);
        
        while (!ct.IsCancellationRequested)
        {
            try
            {
                // 1. 读取 CDC 变更
                var changes = await _cdcReader.ReadChangesAsync(table, lastLsn);
                
                if (changes.Any())
                {
                    // 2. 批量入队
                    foreach (var change in changes)
                    {
                        var task = new SyncTask
                        {
                            TenantId = tenantId,
                            TableName = table,
                            Change = change,
                            EnqueueTime = DateTime.UtcNow,
                            RetryCount = 0
                        };
                        
                        await _queue.Writer.WriteAsync(task, ct);
                        lastLsn = change.Lsn;
                    }
                    
                    // 3. 持久化 LSN
                    await UpdateLastProcessedLsnAsync(tenantId, table, lastLsn);
                }
                else
                {
                    await Task.Delay(500, ct);  // 空轮询间隔
                }
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, $"租户 {tenantId} 表 {table} CDC 监听失败");
                await Task.Delay(5000, ct);  // 失败退避
            }
        }
    }

    private async Task ConsumeLoopAsync(CancellationToken ct, string workerName)
    {
        await foreach (var task in _queue.Reader.ReadAllAsync(ct))
        {
            try
            {
                var startTime = DateTime.UtcNow;
                
                // 1. 获取租户客户端
                using var scope = _serviceProvider.CreateScope();
                var clientFactory = scope.ServiceProvider.GetRequiredService<ITenantClientFactory>();
                var client = clientFactory.GetClient(task.TenantId);
                
                // 2. 路由到对应 Schema
                await UpsertChangeAsync(client, task);
                
                // 3. 记录指标
                var latency = DateTime.UtcNow - task.EnqueueTime;
                _metrics.RecordSyncLatency(task.TenantId, task.TableName, latency);
                _metrics.IncrementSyncSuccess(task.TenantId, task.TableName);
            }
            catch (Exception ex)
            {
                // 4. 重试逻辑
                task.RetryCount++;
                task.LastError = ex.Message;
                
                if (task.RetryCount < 3)
                {
                    // 指数退避后重试
                    await Task.Delay(TimeSpan.FromSeconds(Math.Pow(2, task.RetryCount)), ct);
                    await _queue.Writer.WriteAsync(task, ct);
                    _metrics.IncrementSyncRetry(task.TenantId, task.TableName);
                }
                else
                {
                    // 5. 进入死信队列
                    await _deadLetterQueue.Writer.WriteAsync(task, ct);
                    _metrics.IncrementSyncDlq(task.TenantId, task.TableName);
                    _logger.LogError(ex, $"任务进入死信队列: 租户 {task.TenantId} 表 {task.TableName}");
                }
            }
        }
    }
```

```csharp
public class P2FullSyncService
{
    public async Task FullSyncTenantTableAsync(string tenantId, string sourceTable, string targetTable)
    {
        _logger.LogInformation($"P2 全量同步开始: 租户 {tenantId}, {sourceTable} → {targetTable}");
        var startTime = DateTime.Now;
        
        // 1. 获取源表数据(分批)
        const int batchSize = 10000;
        long totalRows = 0;
        long lastSyncVersion = await GetLastSyncVersionAsync(tenantId, targetTable);
        
        // 2. 流式读取
        await using var sourceConn = new SqlConnection(_sourceConnectionString);
        await sourceConn.OpenAsync();
        
        var sql = $@"
            SELECT * FROM {sourceTable} 
            WHERE F_TENANT_ID = @tenantId 
              AND F_SYNC_VERSION > @lastVersion
            ORDER BY F_SYNC_VERSION
        ";
        
        await using var cmd = new SqlCommand(sql, sourceConn);
        cmd.Parameters.AddWithValue("@tenantId", tenantId);
        cmd.Parameters.AddWithValue("@lastVersion", lastSyncVersion);
        cmd.CommandTimeout = 600;  // 10 分钟超时
        
        await using var reader = await cmd.ExecuteReaderAsync();
        
        // 3. 批量写入分析库
        var analyticsClient = _clientFactory.GetClient(tenantId);
        var batch = new List<DwEntity>();
        
        while (await reader.ReadAsync())
        {
            var entity = MapToEntity(reader);
            batch.Add(entity);
            
            if (batch.Count >= batchSize)
            {
                await analyticsClient.Fastest<DwEntity>().BulkCopyAsync(batch);
                totalRows += batch.Count;
                batch.Clear();
            }
        }
        
        if (batch.Any())
        {
            await analyticsClient.Fastest<DwEntity>().BulkCopyAsync(batch);
            totalRows += batch.Count;
        }
        
        // 4. 更新同步版本
        var maxVersion = await GetMaxSyncVersionAsync(tenantId, sourceTable);
        await UpdateLastSyncVersionAsync(tenantId, targetTable, maxVersion);
        
        var elapsed = DateTime.Now - startTime;
        _logger.LogInformation($"P2 全量同步完成: 租户 {tenantId}, {totalRows} 行, 耗时 {elapsed.TotalMinutes:F1} 分钟");
```

```python
import asyncio
import aiohttp
import time
from concurrent.futures import ThreadPoolExecutor

TENANTS = [f"t{i:03d}" for i in range(1, 21)]
DURATION_MINUTES = 30

async def tenant_load_test(session, tenant_id, duration):
    """单租户负载测试"""
    start = time.time()
    metrics = {"inserted": 0, "errors": 0, "p0_latencies": []}
    
    while time.time() - start < duration * 60:
        try:
            # 1. 注入生产库变更
            test_data = {
                "id": f"LOAD_{tenant_id}_{int(time.time()*1000)}",
                "tenantId": tenant_id,
                "workOrderNo": f"WO_{tenant_id}_{int(time.time())}",
                "workshopId": f"WS_{hash(tenant_id) % 10 + 1}",
                "planQty": 100
            }
            insert_time = time.time()
            
            await session.post("http://prod-db:5001/api/production/workorder", json=test_data)
            
            # 2. 轮询分析库,记录延迟

                        latency = time.time() - insert_time
                        metrics["p0_latencies"].append(latency)
                        metrics["inserted"] += 1
                        break
                await asyncio.sleep(1)
            else:
                metrics["errors"] += 1
        except Exception as e:
            metrics["errors"] += 1
    
    return tenant_id, metrics

async def main():

        tasks = [tenant_load_test(session, t, DURATION_MINUTES) for t in TENANTS]
        results = await asyncio.gather(*tasks)
    
    # 汇总
    total_inserted = sum(r[1]["inserted"] for r in results)
    total_errors = sum(r[1]["errors"] for r in results)
    all_latencies = [l for r in results for l in r[1]["p0_latencies"]]
    
    all_latencies.sort()
    p50 = all_latencies[int(len(all_latencies) * 0.5)]

    p99 = all_latencies[int(len(all_latencies) * 0.99)]
```
