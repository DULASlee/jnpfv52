## 一、目标

在 Claude Code 中实现对 JNPF V5.2 代码库的语义检索。输入自然语言查询（如"动态数据源切换怎么实现的"），自动返回最相关的代码片段及文件路径。

---

## 二、技术选型

| 组件 | 选型 | 理由 |
|------|------|------|
| 嵌入模型 | BAAI/bge-small-zh-v1.5 | 体积小（~100MB），中文优化，归一化向量适合 cosine |
| 向量库 | ChromaDB v1.5.9 | 纯 Python，零配置，持久化本地存储 |
| 接口 | MCP Server（FastMCP） | Claude Code 原生支持，官方推荐的简化写法 |
| 运行时 | Python 3.13.3（已安装） | 满足 >=3.10 要求 |

**资源预估：** 内存 <400MB（模型加载），磁盘 <300MB（模型 + 向量库）

### 步骤 3：编写索引脚本

创建 `.claude/index_code.py`：

```python
# index_code.py - JNPF V5.2 代码语义索引器（修正版 v2.0）
import os
from pathlib import Path
import chromadb
from sentence_transformers import SentenceTransformer

# ========== 配置区 ==========
PROJECT_ROOT = Path(__file__).parent.parent  # D:\JNPF-v52
EXTENSIONS = ['.cs', '.csproj', '.vue', '.ts', '.js', '.json', '.md', '.yaml', '.yml', '.sql']
IGNORE_DIRS = [
    'node_modules', '.git', 'bin', 'obj', 'dist', 'build',
    '.claude', '.vs', '.idea', 'packages', 'Public', 'Library',
]
CHUNK_LINES = 120  # 每块约 120 行（平衡语义完整性与精度）
BATCH_SIZE = 256   # 嵌入批大小（CPU 环境建议 256，GPU 可增大）

# ========== 索引范围 ==========
# 只索引核心源码目录，排除第三方和构建产物
INDEX_DIRS = [
    'backend',      # .NET 后端（~1,960 .cs 文件）
    'docs',         # 文档
    'jnpf-web-vue3/src',       # PC 前端源码
    'jnpf-app-vue3/src',       # 移动端源码
    'jnpf-web-datascreen/src', # 数据大屏源码
]

# ========== 初始化 ==========
print("正在加载嵌入模型...")
model = SentenceTransformer('BAAI/bge-small-zh-v1.5')

client = chromadb.PersistentClient(path=str(Path(__file__).parent / "vector_store"))
collection = client.get_or_create_collection(
    name="jnpf_code",
    configuration={"hnsw": {"space": "cosine"}}  # cosine 距离，适合归一化文本向量
)

# ========== 语义分块 ==========
def split_code(file_path: Path, content: str) -> list[str]:
    lines = content.splitlines()
    if len(lines) <= CHUNK_LINES:
        return [content] if content.strip() else []

    chunks = []
    current_chunk = []
    for line in lines:
        current_chunk.append(line)
        if (len(current_chunk) >= CHUNK_LINES and line.strip() == '') or len(current_chunk) >= CHUNK_LINES * 1.5:
            chunk_text = '\n'.join(current_chunk)
            if chunk_text.strip():
                chunks.append(chunk_text)
            current_chunk = []
    if current_chunk:
        chunk_text = '\n'.join(current_chunk)
        if chunk_text.strip():
            chunks.append(chunk_text)
    return chunks

# ========== 扫描文件 ==========
files_count = 0
chunks_count = 0
all_chunks = []
all_metas = []
all_ids = []

print(f"开始扫描，索引范围: {INDEX_DIRS}")
for index_dir in INDEX_DIRS:
    dir_path = PROJECT_ROOT / index_dir
    if not dir_path.exists():
        print(f"  跳过（不存在）: {index_dir}")
        continue
    for root, dirs, files in os.walk(dir_path):
        dirs[:] = [d for d in dirs if d not in IGNORE_DIRS and not d.startswith('.')]
        for file in files:
            if not any(file.endswith(ext) for ext in EXTENSIONS):
                continue
            file_path = Path(root) / file
            rel_path = file_path.relative_to(PROJECT_ROOT)
            try:
                content = file_path.read_text(encoding='utf-8')
            except (UnicodeDecodeError, PermissionError):
                continue
            chunks = split_code(file_path, content)
            for idx, chunk in enumerate(chunks):
                chunk_id = f"{rel_path}::{idx}"
                all_ids.append(chunk_id)
                all_chunks.append(chunk)
                all_metas.append({
                    "source": str(rel_path).replace('\\', '/'),
                    "chunk_index": idx,
                    "language": file_path.suffix,
                })
                chunks_count += 1
            files_count += 1
            if files_count % 100 == 0:
                print(f"  已扫描 {files_count} 个文件，{chunks_count} 个代码块...")

print(f"\n扫描完成: {files_count} 个文件，{chunks_count} 个代码块")
print("开始生成嵌入向量（首次可能需要较长时间）...")

# ========== 批量嵌入 ==========
for i in range(0, len(all_chunks), BATCH_SIZE):
    batch_chunks = all_chunks[i:i + BATCH_SIZE]
    batch_ids = all_ids[i:i + BATCH_SIZE]
    batch_metas = all_metas[i:i + BATCH_SIZE]
    embeddings = model.encode(batch_chunks, normalize_embeddings=True, show_progress_bar=False).tolist()
    collection.add(
        embeddings=embeddings,
        documents=batch_chunks,
        metadatas=batch_metas,
        ids=batch_ids,
    )
    progress = min(i + BATCH_SIZE, len(all_chunks))
    print(f"  已索引 {progress}/{len(all_chunks)} 块 ({progress * 100 // len(all_chunks)}%)")

print(f"\n索引完成！向量库已保存至 .claude/vector_store/")
print(f"  文件数: {files_count}")
print(f"  代码块数: {chunks_count}")
print(f"  向量维度: {model.get_sentence_embedding_dimension()}")
```

### 步骤 4：编写 MCP 检索服务

创建 `.claude/mcp_search.py`：

```python
# mcp_search.py - JNPF V5.2 语义检索 MCP Server（修正版 v2.0）
import json
from pathlib import Path
from mcp.server.fastmcp import FastMCP
import chromadb
from sentence_transformers import SentenceTransformer

# ========== 初始化 ==========
STORE_PATH = str(Path(__file__).parent / "vector_store")

print("正在加载嵌入模型...", flush=True)
model = SentenceTransformer('BAAI/bge-small-zh-v1.5')
print("正在连接向量库...", flush=True)
client = chromadb.PersistentClient(path=STORE_PATH)
collection = client.get_collection("jnpf_code")
print(f"向量库就绪，共 {collection.count()} 个代码块", flush=True)

# ========== MCP Server ==========
mcp_server = FastMCP("jnpf-vector-search")

@mcp_server.tool()
def search_jnpf_code(query: str, top_k: int = 5) -> str:
    q_emb = model.encode([query], normalize_embeddings=True).tolist()
    results = collection.query(query_embeddings=q_emb, n_results=min(top_k, 20))

    output = []
    for doc, meta, dist in zip(
        results['documents'][0],
        results['metadatas'][0],
        results['distances'][0],
    ):
        similarity = round(1 - dist, 4)
        snippet = doc[:800] + ("..." if len(doc) > 800 else "")
        output.append({
            "source": meta['source'],
            "language": meta.get('language', 'unknown'),
            "similarity": similarity,
            "code_snippet": snippet,
        })

    return json.dumps(output, ensure_ascii=False, indent=2)

@mcp_server.tool()
def vector_db_stats() -> str:
    count = collection.count()
    sample = collection.get(limit=min(count, 1000), include=["metadatas"])
    languages = {}
    sources = set()
    for meta in sample['metadatas']:
        lang = meta.get('language', 'unknown')
        languages[lang] = languages.get(lang, 0) + 1
        sources.add(meta.get('source', ''))
    return json.dumps({
        "total_chunks": count,
        "sampled_files": len(sources),
        "language_distribution": dict(sorted(languages.items(), key=lambda x: -x[1])),
    }, ensure_ascii=False, indent=2)

if __name__ == "__main__":
    mcp_server.run()
```

## 五、与原方案的关键差异

| 项目 | 原方案 | 修正版 | 原因 |
|------|--------|--------|------|
| 文件扩展名 | `.java`, `.xml`, `.yml`, `.properties` | `.cs`, `.vue`, `.ts`, `.js`, `.json`, `.md`, `.sql`, `.csproj`, `.yaml` | JNPF 是 .NET 项目，不是 Java |
| MCP 写法 | 低级 `Server` + 错误 import | `FastMCP`（官方推荐） | 原方案 3 处 import 错误会导致启动崩溃 |
| 距离度量 | 默认 L2 | `cosine` | bge 模型输出归一化向量，cosine 效果更优 |
| 索引范围 | 全项目扫描 | 指定 `INDEX_DIRS`（backend, docs, src） | 排除 node_modules、bin、obj 等非源码目录 |
| 分块策略 | 固定 80 行切割 | 120 行 + 空行处切分 | 保持语义完整性 |
| MCP 配置 | `python`（系统 Python） | 虚拟环境绝对路径 | 确保依赖被加载 |
| 索引时间预估 | 2-10 分钟 | 1-3 小时（全量） | 实际 67,000+ 文件，CPU 编码耗时真实 |
| 额外工具 | 无 | `vector_db_stats` | 方便查看向量库状态 |

## 六、故障排查

| 症状 | 原因 | 解决 |
|------|------|------|
| MCP 工具不显示 | `.mcp.json` 路径错误 | 检查 `command` 是否为 venv 的 python 绝对路径 |
| 检索无结果 | 距离度量不匹配 | 确认 collection 创建时使用了 `cosine`；如已用 L2 创建，需删除 `vector_store/` 重建 |
| 索引时间过长 | 全量扫描文件太多 | 缩小 `INDEX_DIRS` 范围，或增大 `BATCH_SIZE` |
| 内存不足 | 模型 + 大批量嵌入 | 减小 `BATCH_SIZE` 到 128 或 64 |
| 模型下载失败 | 网络问题 | 手动下载 `bge-small-zh-v1.5` 到本地，修改 `SentenceTransformer()` 路径 |
