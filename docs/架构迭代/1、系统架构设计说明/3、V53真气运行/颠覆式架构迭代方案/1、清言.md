### 2.2 生命周期不一致问题
**核心问题代码（SqlSugarConfigureExtensions.cs:43）：**
```csharp
// ISqlSugarClient 注册为 Singleton
services.AddSingleton<ISqlSugarClient>(sqlSugar);
// 但 SqlSugarRepository 使用 Scoped 的 ICacheManager
using var serviceScope = serviceProvider.CreateScope();
var _cacheManager = serviceScope.ServiceProvider.GetService<ICacheManager>();
```
**这是一个严重的生命周期陷阱：**
- Singleton 中的 `sqlSugar` 在并发场景下会被多个租户共享
- Repository 构造函数中创建 `serviceScope` 来获取 Scoped 服务
- **这违反了DI最佳实践：Singleton不应依赖Scoped服务**
**对标ABP vNext：**
```csharp
// ABP vNext —— ISqlSugarRepository 正确的 Scoped 注册
// 每个请求创建新的 ISqlSugarClient 实例
services.AddScoped<ISqlSugarClient>(sp =>
{
    var options = sp.GetRequiredService<IOptions<SqlSugarOptions>>();
    var tenantInfo = sp.GetRequiredService<ITenantInfo>();
    // 每个请求创建独立的数据库连接，避免租户状态泄露
    return new SqlSugarScope(CreateConfig(options, tenantInfo));
});
```
**优化方案：**
```csharp
// 1. 使用 Scoped ISqlSugarClient
services.AddScoped<ISqlSugarClient>(sp =>
{
    var options = sp.GetRequiredService<IOptions<ConnectionStringsOptions>>();
    var httpContext = sp.GetRequiredService<IHttpContextAccessor>().HttpContext;
    var tenantId = httpContext?.User?.FindFirst("TenantId")?.Value;
    return CreateSqlSugarClient(options, tenantId);
});
// 2. 引入DI验证 —— 编译时检查
services.AddTransient<IValidateOptions<ConnectionStringsOptions>, 
    ConnectionStringsOptionsValidator>();
```

**优化方案——引入模块化框架：**
```csharp
// 模块接口定义
public interface IJNPFModule
{
    string ModuleName { get; }
    void ConfigureServices(IServiceCollection services, IConfiguration configuration);
    void OnApplicationInitialization(IServiceProvider serviceProvider);
    int Order { get; }
}
// 系统管理模块
public class SystemsModule : IJNPFModule
{
    public string ModuleName => "Systems";
    public int Order => 10;
    
    public void ConfigureServices(IServiceCollection services, IConfiguration configuration)
    {
        services.AddScoped<IUsersService, UsersService>();
        services.AddScoped<IDepartmentService, DepartmentService>();
        services.AddScoped<IRoleService, RoleService>();
    }
    
    public void OnApplicationInitialization(IServiceProvider serviceProvider)
    {
        // 初始化种子数据等
    }
}
// 模块扫描和注册
public static class ModuleExtensions
{
    public static IServiceCollection AddModules(this IServiceCollection services, 
        IConfiguration configuration)
    {
        var moduleTypes = App.EffectiveTypes
            .Where(t => typeof(IJNPFModule).IsAssignableFrom(t) && !t.IsAbstract);
        
        var modules = moduleTypes
            .Select(Activator.CreateInstance)
            .Cast<IJNPFModule>()
            .OrderBy(m => m.Order)
            .ToList();
        
        foreach (var module in modules)
        {
            module.ConfigureServices(services, configuration);
        }
        
        services.AddSingleton(modules);
        return services;
    }
}
```

public class ProblemDetailsResultProvider : IUnifyResultProvider
{
    public IActionResult OnSucceeded(ActionExecutedContext context, object data)
    {
        var statusCode = context.HttpContext.Response.StatusCode;
        return statusCode switch
        {
            StatusCodes.Status201Created => new CreatedResult(context.HttpContext.Request.Path, data),
            StatusCodes.Status204NoContent => new NoContentResult(),
            _ => new OkObjectResult(new SuccessResponse(data))
        };
    }
    
    public IActionResult OnException(ExceptionContext context, ExceptionMetadata metadata)
    {
        var problemDetails = new ProblemDetails
        {
            Status = metadata.StatusCode,
            Title = metadata.ErrorMessage,
            Detail = metadata.Exception?.StackTrace,
            Instance = context.HttpContext.Request.Path,
            Type = $"https://httpstatuses.com/{metadata.StatusCode}",
            Extensions = { ["traceId"] = context.HttpContext.Items["TraceId"] }
        };
        return new ObjectResult(problemDetails) { StatusCode = metadata.StatusCode };
    }
}

    Task DeleteAsync(TEntity entity, CancellationToken ct = default);

public interface IUnitOfWork : IDisposable

    Task RollbackAsync(CancellationToken ct = default);

// 3. 规约模式 —— DDD核心模式
public abstract class Specification<TEntity>
{
    public abstract Expression<Func<TEntity, bool>> ToExpression();
    
    public static Specification<TEntity> operator &(Specification<TEntity> left, Specification<TEntity> right)
    {
        return new AndSpecification<TEntity>(left, right);
    }
}
// 4. 安全的SqlSugar客户端工厂 —— 每个请求创建独立连接
public class SqlSugarClientFactory : ISqlSugarClientFactory

// 1. 二级缓存 —— 对标 Spring Cache (Caffeine + Redis)
public class TwoLevelCache : ICache
{
    private readonly IMemoryCache _l1Cache;
    private readonly IRedisCache _l2Cache;
    private readonly IEventBus _eventBus;
    
    public async Task<T> GetOrAddAsync<T>(string key, Func<Task<T>> factory, 
        TimeSpan? expiry = null, CancellationToken ct = default)
    {
        // L1: 内存缓存
        if (_l1Cache.TryGetValue(key, out T l1Value))
            return l1Value;
        
        // L2: Redis缓存
        var l2Value = await _l2Cache.GetAsync<T>(key);
        if (l2Value != null)
        {
            _l1Cache.Set(key, l2Value, expiry ?? TimeSpan.FromMinutes(5));
            return l2Value;
        }
        
        // 缓存穿透防护：使用互斥锁防止缓存击穿
        using (await _distributedLock.AcquireAsync($"lock:{key}"))
        {
            // 双重检查
            if (_l1Cache.TryGetValue(key, out l1Value))
                return l1Value;
            
            l2Value = await _l2Cache.GetAsync<T>(key);
            if (l2Value != null)
            {
                _l1Cache.Set(key, l2Value, expiry ?? TimeSpan.FromMinutes(5));
                return l2Value;
            }
            
            // 从数据源获取
            var value = await factory();
            
            if (value != null)
            {
                // 雪崩防护：添加随机TTL偏移
                var ttl = (expiry ?? TimeSpan.FromMinutes(30));
                var jitter = TimeSpan.FromSeconds(Random.Shared.Next(30, 300));
                var actualTtl = ttl + jitter;
                
                _l1Cache.Set(key, value, TimeSpan.FromMinutes(5));
                await _l2Cache.SetAsync(key, value, actualLtl);
            }
            else
            {
                // 穿透防护：缓存空值
                _l1Cache.Set(key, default(T), TimeSpan.FromMinutes(1));
                await _l2Cache.SetAsync(key, default(T), TimeSpan.FromMinutes(1));
            }
            
            return value;
        }
    }
}
