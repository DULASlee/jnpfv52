### 0.1 JSON Schema 样本提取

**目标：拿到现有的表单配置和大屏配置的真实 JSON 结构，这是定义 IR 的原材料。**

```bash
# ============================================================
# 这一步需要工程师手动操作 + 脚本辅助
# ============================================================

# --- 表单 Schema ---
# 操作步骤：
#   1. 浏览器打开 JNPF 平台
#   2. 进入"表单设计器"
#   3. 打开 F12 → Network 面板
#   4. 打开一个已有的复杂表单配置（如"用户编辑表单"、"订单表单"、"审批表单"）
#   5. 找到加载配置的 API 响应（通常是 GET 请求，返回大段 JSON）
#   6. 右键 → Copy Response → 保存到文件
#
# 至少需要 3 个复杂度不同的表单配置：
#   简单表单（5-10个字段，无联动）
#   中等表单（15-20个字段，有联动和条件显隐）
#   复杂表单（30+字段，有联动、校验、计算字段、子表）
#
# 保存路径：
#   ~/jnpf-survey/phase0/schema-form-simple.json
#   ~/jnpf-survey/phase0/schema-form-medium.json
#   ~/jnpf-survey/phase0/schema-form-complex.json

# --- 大屏 Schema ---
# 操作步骤：
#   1. 进入"大屏设计器"
#   2. 打开 F12 → Network
#   3. 打开一个已有的复杂大屏
#   4. 找到加载配置的 API 响应
#   5. 保存
#
# 至少需要 2 个：
#   ~/jnpf-survey/phase0/schema-screen-simple.json
#   ~/jnpf-survey/phase0/schema-screen-complex.json

# --- 详情页/列表页 Schema ---
# 同理，如果列表页和详情页也有配置化，也需要提取
#   ~/jnpf-survey/phase0/schema-list.json
#   ~/jnpf-survey/phase0/schema-detail.json

# ============================================================
# 拿到 Schema 后，用脚本分析其结构
# ============================================================

# 1. 分析 JSON 的顶层结构（有哪些 key）
echo "===== 表单 Schema 顶层结构 =====" > ~/jnpf-survey/phase0/schema-analysis.txt
for f in ~/jnpf-survey/phase0/schema-form-*.json; do
  echo "--- $f ---" >> ~/jnpf-survey/phase0/schema-analysis.txt
  # 提取前3层 key 结构
  python3 -c "
import json, sys
def show_keys(obj, prefix='', depth=0):
    if depth > 3: return
    if isinstance(obj, dict):
        for k, v in obj.items():
            vtype = type(v).__name__
            if isinstance(v, (dict, list)):
                vtype = f'{type(v).__name__}({len(v)})'
            print(f'{prefix}{k}: {vtype}')
            show_keys(v, prefix + '  ', depth + 1)
    elif isinstance(obj, list) and len(obj) > 0:
        print(f'{prefix}[0]: (list item)')
        show_keys(obj[0], prefix + '  ', depth + 1)
with open('$f') as fh:
    data = json.load(fh)
    show_keys(data)
" >> ~/jnpf-survey/phase0/schema-analysis.txt 2>/dev/null
  echo "" >> ~/jnpf-survey/phase0/schema-analysis.txt
done

# 2. 统计表单 Schema 中的字段数量
echo "===== 字段统计 =====" >> ~/jnpf-survey/phase0/schema-analysis.txt
for f in ~/jnpf-survey/phase0/schema-form-*.json; do
  echo "--- $f ---" >> ~/jnpf-survey/phase0/schema-analysis.txt
  python3 -c "
import json
with open('$f') as fh:
    data = json.load(fh)
# 尝试不同的 Schema 结构路径
for key in ['fields', 'columns', 'formItems', 'widgetList', 'list', 'data', 'formData']:
    if key in data:
        items = data[key]
        if isinstance(items, list):
            print(f'  {key}: {len(items)} items')
            if len(items) > 0 and isinstance(items[0], dict):
                print(f'  [0] keys: {list(items[0].keys())[:15]}')
" >> ~/jnpf-survey/phase0/schema-analysis.txt 2>/dev/null
done

# 3. 提取所有表达式（visible / disabled / rule / formatter 等）
echo "===== 表达式提取 =====" > ~/jnpf-survey/phase0/expressions-extracted.txt

python3 << 'PYEOF' >> ~/jnpf-survey/phase0/expressions-extracted.txt
import json, os, re

EXPR_KEYS = ['visible', 'disabled', 'required', 'rule', 'rules', 'formatter',
             'formula', 'expression', 'condition', 'validator', 'onChange',
             'onVisible', 'computedValue', 'defaultExpression', 'linkage',
             'visibleOn', 'disabledOn', 'requiredOn', 'hidden', 'readOnly']

def extract_expressions(obj, path=""):
    results = []
    if isinstance(obj, dict):
        for k, v in obj.items():
            current_path = f"{path}.{k}" if path else k
            if k.lower() in [ek.lower() for ek in EXPR_KEYS]:
                if isinstance(v, str) and len(v) > 2:
                    results.append((current_path, v))
                elif isinstance(v, list):
                    for i, item in enumerate(v):
                        if isinstance(item, str):
                            results.append((f"{current_path}[{i}]", item))
            results.extend(extract_expressions(v, current_path))
    elif isinstance(obj, list):
        for i, item in enumerate(obj):
            results.extend(extract_expressions(item, f"{path}[{i}]"))
    return results

for fname in sorted(os.listdir(os.path.expanduser("~/jnpf-survey/phase0/"))):
    if fname.endswith(".json"):
        fpath = os.path.expanduser(f"~/jnpf-survey/phase0/{fname}")
        try:
            with open(fpath) as f:
                data = json.load(f)
            exprs = extract_expressions(data)
            print(f"\n--- {fname} ({len(exprs)} expressions) ---")
            for path, expr in exprs[:50]:  # 最多显示50个
                preview = expr[:200] + "..." if len(expr) > 200 else expr
                print(f"  [{path}] = {repr(preview)}")
            if len(exprs) > 50:
                print(f"  ... 还有 {len(exprs) - 50} 个表达式")
        except Exception as e:
            print(f"\n--- {fname}: 解析失败: {e} ---")
PYEOF

# 4. 扫描 Schema 中是否混入了函数/闭包（这是 IR 的死穴）
echo "===== 函数/闭包检测 =====" >> ~/jnpf-survey/phase0/expressions-extracted.txt

python3 << 'PYEOF' >> ~/jnpf-survey/phase0/expressions-extracted.txt
import json, os

DANGER_PATTERNS = [
    'function(', 'function (', '=>', 'return ', 'this.',
    'window.', 'document.', '$store', '$parent', '$root',
    'async ', 'await ', 'import(', 'require(',
    'setTimeout', 'setInterval', 'console.',
    'new Function', 'eval(', '__proto__', 'constructor[',
]

for fname in sorted(os.listdir(os.path.expanduser("~/jnpf-survey/phase0/"))):
    if not fname.endswith(".json"):
        continue
    fpath = os.path.expanduser(f"~/jnpf-survey/phase0/{fname}")
    try:
        with open(fpath) as f:
            content = f.read()
        
        print(f"\n--- {fname} 危险模式检测 ---")
        for pattern in DANGER_PATTERNS:
            count = content.count(pattern)
            if count > 0:
                print(f"  ⚠️ '{pattern}' 出现 {count} 次")
    except:
        pass
PYEOF

# 5. 输出最终分析摘要
echo ""
echo "===== Schema 分析完成 ====="
echo "请查看："
echo "  ~/jnpf-survey/phase0/schema-analysis.txt    — JSON 结构分析"
echo "  ~/jnpf-survey/phase0/expressions-extracted.txt — 表达式提取 + 危险模式检测"
```

### 0.4 现有代码生成产物解剖

**目标：拿到现有代码生成器的产物，分析"85%重复"到底重复在哪、平台锁定度有多高。**

```bash
# ============================================================
# 这一步需要工程师手动操作
# ============================================================
#
# 操作步骤：
#   1. 浏览器打开 JNPF 平台
#   2. 进入"代码生成器"
#   3. 选择一个已有的业务模块（如"用户管理"或"订单管理"）
#   4. 点击"生成代码" / "下载代码"
#   5. 得到一个 ZIP 包
#   6. 解压到 ~/jnpf-survey/phase0/generated-project/
#
# 至少生成 2 套：
#   简单的（CRUD，无复杂逻辑）
#   复杂的（有工作流、有子表、有自定义按钮）
#
# 解压后执行以下分析：

GENERATED_DIR=~/jnpf-survey/phase0/generated-project

# 检查是否已有生成代码
if [ -d "$GENERATED_DIR" ]; then

  # ============================================================
  # 1. 代码量统计
  # ============================================================
  echo "===== 生成代码量统计 =====" > ~/jnpf-survey/phase0/generated-analysis.txt
  
  echo "--- 文件列表 ---" >> ~/jnpf-survey/phase0/generated-analysis.txt
  find "$GENERATED_DIR" -name "*.vue" -o -name "*.js" -o -name "*.ts" | sort >> ~/jnpf-survey/phase0/generated-analysis.txt
  
  echo "" >> ~/jnpf-survey/phase0/generated-analysis.txt
  echo "--- 行数统计 ---" >> ~/jnpf-survey/phase0/generated-analysis.txt
  find "$GENERATED_DIR" -name "*.vue" -o -name "*.js" -o -name "*.ts" | xargs wc -l 2>/dev/null >> ~/jnpf-survey/phase0/generated-analysis.txt
  
  # ============================================================
  # 2. 对 JNPF 内部库的依赖度（平台锁定分析）
  # ============================================================
  echo "" >> ~/jnpf-survey/phase0/generated-analysis.txt
  echo "--- 平台依赖度分析 ---" >> ~/jnpf-survey/phase0/generated-analysis.txt
  
  echo "import @/components/jnpf: $(grep -rn 'from.*@/components/jnpf\|from.*jnpf' "$GENERATED_DIR" | wc -l)" >> ~/jnpf-survey/phase0/generated-analysis.txt
  echo "import @/api/: $(grep -rn 'from.*@/api/' "$GENERATED_DIR" | wc -l)" >> ~/jnpf-survey/phase0/generated-analysis.txt
  echo "import @/utils/: $(grep -rn 'from.*@/utils/' "$GENERATED_DIR" | wc -l)" >> ~/jnpf-survey/phase0/generated-analysis.txt
  echo "import @/store/: $(grep -rn 'from.*@/store/' "$GENERATED_DIR" | wc -l)" >> ~/jnpf-survey/phase0/generated-analysis.txt
  echo "import element-plus: $(grep -rn 'from.*element-plus' "$GENERATED_DIR" | wc -l)" >> ~/jnpf-survey/phase0/generated-analysis.txt
  echo "" >> ~/jnpf-survey/phase0/generated-analysis.txt
  
  echo "--- 详细依赖列表 ---" >> ~/jnpf-survey/phase0/generated-analysis.txt
  grep -rn "from '@/\|from \"@/" "$GENERATED_DIR" | sort -u >> ~/jnpf-survey/phase0/generated-analysis.txt
  
  # ============================================================
  # 3. 代码重复度分析
  # ============================================================
  echo "" >> ~/jnpf-survey/phase0/generated-analysis.txt
  echo "--- 代码重复度分析 ---" >> ~/jnpf-survey/phase0/generated-analysis.txt
  
  # 提取所有 .vue 文件的 <script> 部分，对比相似度
  for f in $(find "$GENERATED_DIR" -name "*.vue"); do
    echo "### $f ###" >> ~/jnpf-survey/phase0/generated-analysis.txt
    # 提取 script 部分的行数
    script_lines=$(sed -n '/<script/,/<\/script>/p' "$f" | wc -l)
    template_lines=$(sed -n '/<template/,/<\/template>/p' "$f" | wc -l)
    total_lines=$(wc -l < "$f")
    echo "  total: $total_lines, template: $template_lines, script: $script_lines" >> ~/jnpf-survey/phase0/generated-analysis.txt
  done
  
  # ============================================================
  # 4. 生成代码中是否用了 eval
  # ============================================================
  echo "" >> ~/jnpf-survey/phase0/generated-analysis.txt
  echo "--- 生成代码中的 eval ---" >> ~/jnpf-survey/phase0/generated-analysis.txt
  grep -rn "eval\|new Function" "$GENERATED_DIR" >> ~/jnpf-survey/phase0/generated-analysis.txt
  
  # ============================================================
  # 5. 生成代码的结构分层
  # ============================================================
  echo "" >> ~/jnpf-survey/phase0/generated-analysis.txt
```
